#include "md_knn.h"

void md_kernel(
    TYPE force_x[nAtoms],
    TYPE force_y[nAtoms],
    TYPE force_z[nAtoms],
    TYPE position_x[nAtoms],
    TYPE position_y[nAtoms],
    TYPE position_z[nAtoms],
    int32_t NL[nAtoms * maxNeighbors]) {

    // Iterate over each atom
    loop_i:
    for (int i = 0; i < nAtoms; i++) {
        TYPE fx = 0.0;
        TYPE fy = 0.0;
        TYPE fz = 0.0;

        // Get current atom's position
        TYPE xi = position_x[i];
        TYPE yi = position_y[i];
        TYPE zi = position_z[i];

        // Iterate over each neighbor of the current atom
        loop_j:
        for (int j = 0; j < maxNeighbors; j++) {
            int neighbor_index = NL[i * maxNeighbors + j];

            // Check if the neighbor index is valid (not -1)
            if (neighbor_index != -1) {
                // Get neighbor's position
                TYPE xj = position_x[neighbor_index];
                TYPE yj = position_y[neighbor_index];
                TYPE zj = position_z[neighbor_index];

                // Calculate distance components
                TYPE dx = xi - xj;
                TYPE dy = yi - yj;
                TYPE dz = zi - zj;

                // Calculate squared distance
                TYPE r2 = dx * dx + dy * dy + dz * dz;

                // Avoid division by zero
                if (r2 > 0.0) {
                    // Calculate r^2 and r^6
                    TYPE r2_inv = 1.0 / r2;
                    TYPE r6_inv = r2_inv * r2_inv * r2_inv;

                    // Calculate force magnitude using Lennard-Jones potential
                    TYPE force_mag = (lj1 * r6_inv - lj2) * r6_inv * r2_inv;

                    // Calculate force components
                    TYPE fx_ij = dx * force_mag;
                    TYPE fy_ij = dy * force_mag;
                    TYPE fz_ij = dz * force_mag;

                    // Accumulate forces
                    fx += fx_ij;
                    fy += fy_ij;
                    fz += fz_ij;
                }
            }
        }

        // Update force arrays
        force_x[i] = fx;
        force_y[i] = fy;
        force_z[i] = fz;
    }
}
