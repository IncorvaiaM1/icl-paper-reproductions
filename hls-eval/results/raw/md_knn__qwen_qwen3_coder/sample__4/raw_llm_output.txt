#include "md_knn.h"
#include <math.h>

void md_kernel(
    TYPE force_x[nAtoms],
    TYPE force_y[nAtoms],
    TYPE force_z[nAtoms],
    TYPE position_x[nAtoms],
    TYPE position_y[nAtoms],
    TYPE position_z[nAtoms],
    int32_t NL[nAtoms * maxNeighbors]) {

    // Initialize forces to zero
    for (int i = 0; i < nAtoms; i++) {
        force_x[i] = 0.0;
        force_y[i] = 0.0;
        force_z[i] = 0.0;
    }

    // Outer loop over all atoms
    loop_i: for (int i = 0; i < nAtoms; i++) {
        TYPE i_pos_x = position_x[i];
        TYPE i_pos_y = position_y[i];
        TYPE i_pos_z = position_z[i];

        // Inner loop over neighbors of atom i
        loop_j: for (int j_idx = 0; j_idx < maxNeighbors; j_idx++) {
            int j = NL[i * maxNeighbors + j_idx];

            // Check for invalid neighbor index (padding)
            if (j < 0 || j >= nAtoms) continue;

            // Compute distance vector
            TYPE dx = position_x[j] - i_pos_x;
            TYPE dy = position_y[j] - i_pos_y;
            TYPE dz = position_z[j] - i_pos_z;

            // Compute squared distance
            TYPE r2 = dx * dx + dy * dy + dz * dz;

            // Avoid division by zero
            if (r2 > 0.0) {
                // Compute r^2, r^6, and r^12
                TYPE r2_inv = 1.0 / r2;
                TYPE r6_inv = r2_inv * r2_inv * r2_inv;
                TYPE r12_inv = r6_inv * r6_inv;

                // Compute force magnitude using Lennard-Jones potential
                TYPE force_mag = (12.0 * lj1 * r12_inv - 6.0 * lj2 * r6_inv) * r2_inv;

                // Compute force components
                TYPE fx = force_mag * dx;
                TYPE fy = force_mag * dy;
                TYPE fz = force_mag * dz;

                // Update forces (Newton's third law: equal and opposite forces)
                force_x[i] += fx;
                force_y[i] += fy;
                force_z[i] += fz;
                force_x[j] -= fx;
                force_y[j] -= fy;
                force_z[j] -= fz;
            }
        }
    }
}
