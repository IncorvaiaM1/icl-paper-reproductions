#include "covariance.h"

void kernel_covariance(double float_n, double data[32][28], double cov[28][28], double mean[28]) {
    // Compute mean for each attribute
    for (int j = 0; j < 28; j++) {
        mean[j] = 0.0;
        for (int i = 0; i < 32; i++) {
            mean[j] += data[i][j];
        }
        mean[j] /= float_n;
    }

    // Compute covariance matrix
    for (int i = 0; i < 28; i++) {
        for (int j = 0; j < 28; j++) {
            double sum = 0.0;
            for (int k = 0; k < 32; k++) {
                double dev_i = data[k][i] - mean[i];
                double dev_j = data[k][j] - mean[j];
                sum += dev_i * dev_j;
            }
            cov[i][j] = sum / (float_n - 1.0);
        }
    }
}
