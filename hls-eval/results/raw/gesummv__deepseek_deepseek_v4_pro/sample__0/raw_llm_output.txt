#include "gesummv.h"

void kernel_gesummv(
    double alpha,
    double beta,
    double A[30][30],
    double B[30][30],
    double tmp[30],
    double x[30],
    double y[30]) {
    for (int i = 0; i < 30; i++) {
        double tmp_i = 0.0;
        double y_i = 0.0;

        for (int j = 0; j < 30; j++) {
            tmp_i += A[i][j] * x[j];
            y_i += B[i][j] * x[j];
        }

        tmp[i] = tmp_i;
        y[i] = alpha * tmp_i + beta * y_i;
    }
}
