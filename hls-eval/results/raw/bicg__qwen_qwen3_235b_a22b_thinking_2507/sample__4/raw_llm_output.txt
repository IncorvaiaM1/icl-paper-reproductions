    // Compute q = A * p
    for (int i = 0; i < 42; ++i) {
        q[i] = 0.0;

            q[i] += A[i][j] * p[j];

    for (int j = 0; j < 38; ++j) {
        s[j] = 0.0;

        }
    }
}
