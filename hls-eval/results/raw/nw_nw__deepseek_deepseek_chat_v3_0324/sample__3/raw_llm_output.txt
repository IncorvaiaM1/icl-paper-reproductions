#include "nw_nw.h"

#define MATCH_SCORE 1
#define MISMATCH_SCORE -1
#define GAP_SCORE -1

void needwun(
    char SEQA[ALEN],
    char SEQB[BLEN],
    char alignedA[ALEN + BLEN],
    char alignedB[ALEN + BLEN],
    int M[(ALEN + 1) * (BLEN + 1)],
    char ptr[(ALEN + 1) * (BLEN + 1)]) {

    // Initialize the first row and column of the scoring matrix
    for (int i = 0; i <= ALEN; i++) {
        M[i * (BLEN + 1)] = i * GAP_SCORE;
        ptr[i * (BLEN + 1)] = 'U';
    }
    for (int j = 0; j <= BLEN; j++) {
        M[j] = j * GAP_SCORE;
        ptr[j] = 'L';
    }
    ptr[0] = 'D';

    // Fill the scoring matrix
    for (int i = 1; i <= ALEN; i++) {
        for (int j = 1; j <= BLEN; j++) {
            int match = M[(i - 1) * (BLEN + 1) + (j - 1)] + 
                       (SEQA[i - 1] == SEQB[j - 1] ? MATCH_SCORE : MISMATCH_SCORE);
            int del = M[(i - 1) * (BLEN + 1) + j] + GAP_SCORE;
            int ins = M[i * (BLEN + 1) + (j - 1)] + GAP_SCORE;

            if (match >= del && match >= ins) {
                M[i * (BLEN + 1) + j] = match;
                ptr[i * (BLEN + 1) + j] = 'D';
            } else if (del >= ins) {
                M[i * (BLEN + 1) + j] = del;
                ptr[i * (BLEN + 1) + j] = 'U';
            } else {
                M[i * (BLEN + 1) + j] = ins;
                ptr[i * (BLEN + 1) + j] = 'L';
            }
        }
    }

    // Traceback to find the optimal alignment
    int i = ALEN;
    int j = BLEN;
    int pos = ALEN + BLEN - 1;

    while (i > 0 || j > 0) {
        if (ptr[i * (BLEN + 1) + j] == 'D') {
            alignedA[pos] = SEQA[i - 1];
            alignedB[pos] = SEQB[j - 1];
            i--;
            j--;
        } else if (ptr[i * (BLEN + 1) + j] == 'U') {
            alignedA[pos] = SEQA[i - 1];
            alignedB[pos] = '_';
            i--;
        } else {
            alignedA[pos] = '_';
            alignedB[pos] = SEQB[j - 1];
            j--;
        }
        pos--;
    }

    // Pad the remaining positions with underscores
    while (pos >= 0) {
        alignedA[pos] = '_';
        alignedB[pos] = '_';
        pos--;
    }
}
