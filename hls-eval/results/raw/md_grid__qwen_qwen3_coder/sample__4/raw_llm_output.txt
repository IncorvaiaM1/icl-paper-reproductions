#include "md_grid.h"

void md(
    int32_t n_points[blockSide][blockSide][blockSide],
    dvector_t force[blockSide][blockSide][blockSide][densityFactor],
    dvector_t position[blockSide][blockSide][blockSide][densityFactor]) {

    // Zero out force array
    loop_init_force:
    for (int i = 0; i < blockSide; i++) {
        for (int j = 0; j < blockSide; j++) {
            for (int k = 0; k < blockSide; k++) {
                for (int d = 0; d < densityFactor; d++) {
                    force[i][j][k][d].x = 0.0;
                    force[i][j][k][d].y = 0.0;
                    force[i][j][k][d].z = 0.0;
                }
            }
        }
    }

    // Iterate over each block in the grid
    loop_grid0:
    for (int a = 0; a < blockSide; a++) {
        for (int b = 0; b < blockSide; b++) {
            for (int c = 0; c < blockSide; c++) {
                // Iterate over a 3x3x3 cube of blocks around (a,b,c)
                loop_grid1:
                for (int nei_x = -1; nei_x <= 1; nei_x++) {
                    for (int nei_y = -1; nei_y <= 1; nei_y++) {
                        for (int nei_z = -1; nei_z <= 1; nei_z++) {
                            // Calculate neighbor block indices with wrapping
                            int nbx = (a + nei_x + blockSide) % blockSide;
                            int nby = (b + nei_y + blockSide) % blockSide;
                            int nbz = (c + nei_z + blockSide) % blockSide;

                            // Iterate over points in current block (a,b,c)
                            loop_p:
                            for (int p = 0; p < n_points[a][b][c]; p++) {
                                // Iterate over points in neighbor block (nbx,nby,nbz)
                                loop_q:
                                for (int q = 0; q < n_points[nbx][nby][nbz]; q++) {
                                    // Skip if same point when blocks are identical
                                    if ((nei_x == 0) && (nei_y == 0) && (nei_z == 0) && (p == q))
                                        continue;

                                    // Compute distance vector
                                    TYPE dx = position[a][b][c][p].x - position[nbx][nby][nbz][q].x;
                                    TYPE dy = position[a][b][c][p].y - position[nbx][nby][nbz][q].y;
                                    TYPE dz = position[a][b][c][p].z - position[nbx][nby][nbz][q].z;

                                    // Compute squared distance
                                    TYPE r2 = dx*dx + dy*dy + dz*dz;

                                    // Skip if too close to avoid numerical issues
                                    if (r2 < 1e-12)
                                        continue;

                                    // Compute r^-2
                                    TYPE r2_inv = 1.0 / r2;

                                    // Compute r^-6
                                    TYPE r6_inv = r2_inv * r2_inv * r2_inv;

                                    // Compute force magnitude: F(r) = 6 * r^-7 * (lj1 * r^-6 - lj2) + 6 * lj1 * r^-13
                                    TYPE force_mag = 6.0 * r2_inv * r6_inv * (lj1 * r6_inv - lj2) + 6.0 * lj1 * r6_inv * r6_inv * r2_inv;

                                    // Compute force components
                                    TYPE fx = force_mag * dx;
                                    TYPE fy = force_mag * dy;
                                    TYPE fz = force_mag * dz;

                                    // Update forces on both points (Newton's third law)
                                    force[a][b][c][p].x += fx;
                                    force[a][b][c][p].y += fy;
                                    force[a][b][c][p].z += fz;

                                    force[nbx][nby][nbz][q].x -= fx;
                                    force[nbx][nby][nbz][q].y -= fy;
                                    force[nbx][nby][nbz][q].z -= fz;
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
