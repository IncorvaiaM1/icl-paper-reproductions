#include "md_grid.h"

void md(
    int32_t n_points[blockSide][blockSide][blockSide],
    dvector_t force[blockSide][blockSide][blockSide][densityFactor],
    dvector_t position[blockSide][blockSide][blockSide][densityFactor]) {

    // Initialize forces to zero
    for (int i = 0; i < blockSide; i++) {
        for (int j = 0; j < blockSide; j++) {
            for (int k = 0; k < blockSide; k++) {
                for (int d = 0; d < densityFactor; d++) {
                    force[i][j][k][d].x = 0.0;
                    force[i][j][k][d].y = 0.0;
                    force[i][j][k][d].z = 0.0;
                }
            }
        }
    }

    // Iterate over each block in the grid
    for (int bx = 0; bx < blockSide; bx++) {
        for (int by = 0; by < blockSide; by++) {
            for (int bz = 0; bz < blockSide; bz++) {
                // Iterate over the 3x3x3 neighborhood of blocks
                for (int dx = -1; dx <= 1; dx++) {
                    for (int dy = -1; dy <= 1; dy++) {
                        for (int dz = -1; dz <= 1; dz++) {
                            // Calculate neighbor block indices with wrapping
                            int nx = (bx + dx + blockSide) % blockSide;
                            int ny = (by + dy + blockSide) % blockSide;
                            int nz = (bz + dz + blockSide) % blockSide;

                            // Iterate over points in the current block
                            for (int p = 0; p < n_points[bx][by][bz]; p++) {
                                // Load position of point p
                                TYPE px = position[bx][by][bz][p].x;
                                TYPE py = position[bx][by][bz][p].y;
                                TYPE pz = position[bx][by][bz][p].z;

                                // Iterate over points in the neighbor block
                                for (int q = 0; q < n_points[nx][ny][nz]; q++) {
                                    // Load position of point q
                                    TYPE qx = position[nx][ny][nz][q].x;
                                    TYPE qy = position[nx][ny][nz][q].y;
                                    TYPE qz = position[nx][ny][nz][q].z;

                                    // Calculate distance vector
                                    TYPE dx_vec = px - qx;
                                    TYPE dy_vec = py - qy;
                                    TYPE dz_vec = pz - qz;

                                    // Calculate squared distance
                                    TYPE r2 = dx_vec * dx_vec + dy_vec * dy_vec + dz_vec * dz_vec;

                                    // Avoid division by zero
                                    if (r2 > 1e-12) {
                                        // Calculate r^-2
                                        TYPE r2_inv = 1.0 / r2;
                                        // Calculate r^-6
                                        TYPE r6_inv = r2_inv * r2_inv * r2_inv;
                                        // Calculate r^-7
                                        TYPE r7_inv = r6_inv * r2_inv;
                                        // Calculate r^-13
                                        TYPE r13_inv = r6_inv * r7_inv;

                                        // Calculate force magnitude
                                        TYPE f_mag = 6.0 * r7_inv * (lj1 * r6_inv - lj2) + 6.0 * lj1 * r13_inv;

                                        // Calculate force components
                                        TYPE fx = f_mag * dx_vec;
                                        TYPE fy = f_mag * dy_vec;
                                        TYPE fz = f_mag * dz_vec;

                                        // Update forces (Newton's third law)
                                        force[bx][by][bz][p].x += fx;
                                        force[bx][by][bz][p].y += fy;
                                        force[bx][by][bz][p].z += fz;
                                        force[nx][ny][nz][q].x -= fx;
                                        force[nx][ny][nz][q].y -= fy;
                                        force[nx][ny][nz][q].z -= fz;
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
