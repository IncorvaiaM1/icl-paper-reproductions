#include "md_grid.h"

void md(
    int32_t n_points[blockSide][blockSide][blockSide],
    dvector_t force[blockSide][blockSide][blockSide][densityFactor],
    dvector_t position[blockSide][blockSide][blockSide][densityFactor]) {
    
    // Zero out force array
    for (int i = 0; i < blockSide; i++) {
        for (int j = 0; j < blockSide; j++) {
            for (int k = 0; k < blockSide; k++) {
                for (int d = 0; d < densityFactor; d++) {
                    force[i][j][k][d].x = 0.0;
                    force[i][j][k][d].y = 0.0;
                    force[i][j][k][d].z = 0.0;
                }
            }
        }
    }

    // Loop over all blocks in the grid
    loop_grid0:
    for (int i = 0; i < blockSide; i++) {
        for (int j = 0; j < blockSide; j++) {
            for (int k = 0; k < blockSide; k++) {
                
                // Loop over the 3x3x3 neighborhood of blocks
                loop_grid1:
                for (int oi = -1; oi <= 1; oi++) {
                    for (int oj = -1; oj <= 1; oj++) {
                        for (int ok = -1; ok <= 1; ok++) {
                            
                            // Calculate neighbor block indices with wrapping
                            int ni = (i + oi + blockSide) % blockSide;
                            int nj = (j + oj + blockSide) % blockSide;
                            int nk = (k + ok + blockSide) % blockSide;
                            
                            // Loop over points in current block
                            loop_p:
                            for (int p = 0; p < n_points[i][j][k]; p++) {
                                
                                // Loop over points in neighbor block
                                loop_q:
                                for (int q = 0; q < n_points[ni][nj][nk]; q++) {
                                    
                                    // Calculate distance vector
                                    dvector_t dr;
                                    dr.x = position[i][j][k][p].x - position[ni][nj][nk][q].x;
                                    dr.y = position[i][j][k][p].y - position[ni][nj][nk][q].y;
                                    dr.z = position[i][j][k][p].z - position[ni][nj][nk][q].z;
                                    
                                    // Calculate squared distance
                                    TYPE dr2 = dr.x * dr.x + dr.y * dr.y + dr.z * dr.z;
                                    
                                    // Skip if same point
                                    if (dr2 == 0.0) continue;
                                    
                                    // Calculate r^-2
                                    TYPE inv_dr2 = 1.0 / dr2;
                                    
                                    // Calculate r^-6
                                    TYPE inv_dr6 = inv_dr2 * inv_dr2 * inv_dr2;
                                    
                                    // Calculate r^-7 and r^-13
                                    TYPE inv_dr7 = inv_dr6 * inv_dr2;
                                    TYPE inv_dr13 = inv_dr6 * inv_dr7;
                                    
                                    // Calculate force magnitude using the derivative of LJ potential
                                    TYPE force_mag = 6.0 * inv_dr7 * (lj1 * inv_dr6 - lj2) + 6.0 * lj1 * inv_dr13;
                                    
                                    // Calculate force components
                                    dvector_t f;
                                    f.x = dr.x * force_mag;
                                    f.y = dr.y * force_mag;
                                    f.z = dr.z * force_mag;
                                    
                                    // Update forces (add to current block point, subtract from neighbor block point)
                                    force[i][j][k][p].x += f.x;
                                    force[i][j][k][p].y += f.y;
                                    force[i][j][k][p].z += f.z;
                                    
                                    force[ni][nj][nk][q].x -= f.x;
                                    force[ni][nj][nk][q].y -= f.y;
                                    force[ni][nj][nk][q].z -= f.z;
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
