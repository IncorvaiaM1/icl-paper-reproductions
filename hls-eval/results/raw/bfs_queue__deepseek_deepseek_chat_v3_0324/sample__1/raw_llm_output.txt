#include "bfs_queue.h"

void bfs(
    node_t nodes[N_NODES],
    edge_t edges[N_EDGES],
    node_index_t starting_node,
    level_t level[N_NODES],
    edge_index_t level_counts[N_LEVELS]
) {
    // Initialize levels to MAX_LEVEL (unvisited)
    for (node_index_t i = 0; i < N_NODES; i++) {
        level[i] = MAX_LEVEL;
    }
    level[starting_node] = 0;
    level_counts[0] = 1; // Starting node is at level 0

    bool done = false;
    level_t current_level = 0;

    while (!done && current_level < N_LEVELS - 1) {
        edge_index_t count = 0;

        // Process all nodes at current_level
        for (node_index_t i = 0; i < N_NODES; i++) {
            if (level[i] == current_level) {
                // Explore all neighbors of node i
                edge_index_t edge_begin = nodes[i].edge_begin;
                edge_index_t edge_end = nodes[i].edge_end;

                for (edge_index_t j = edge_begin; j < edge_end; j++) {
                    node_index_t neighbor = edges[j].dst;

                    if (level[neighbor] == MAX_LEVEL) {
                        level[neighbor] = current_level + 1;
                        count++;
                    }
                }
            }
        }

        level_counts[current_level + 1] = count;
        current_level++;

        if (count == 0) {
            done = true;
        }
    }
}
