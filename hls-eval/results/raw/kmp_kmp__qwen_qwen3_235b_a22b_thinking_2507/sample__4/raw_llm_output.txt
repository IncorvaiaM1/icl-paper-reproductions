#include "kmp_kmp.h"

static void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    kmpNext[0] = 0;
    for (int i = 1; i < PATTERN_SIZE; i++) {
        int j = kmpNext[i-1];
        while (j > 0 && pattern[i] != pattern[j]) {
            j = kmpNext[j-1];
        }
        if (pattern[i] == pattern[j]) {
            j++;
        }
        kmpNext[i] = j;
    }
}

int kmp(char pattern[PATTERN_SIZE], char input[STRING_SIZE], int32_t kmpNext[PATTERN_SIZE], int32_t n_matches[1]) {
    CPF(pattern, kmpNext);
    
    int j = 0;
    int count = 0;
    
    for (int i = 0; i < STRING_SIZE; i++) {
        while (j > 0 && input[i] != pattern[j]) {
            j = kmpNext[j-1];
        }
        if (input[i] == pattern[j]) {
            j++;
        }
        if (j == PATTERN_SIZE) {
            count++;
            j = kmpNext[j-1];
        }
    }
    
    n_matches[0] = count;
    return count;
}
