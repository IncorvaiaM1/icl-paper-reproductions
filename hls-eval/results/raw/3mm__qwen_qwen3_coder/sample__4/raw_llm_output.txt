#include "3mm.h"

void kernel_3mm(
    double E[16][18],
    double A[16][20],
    double B[20][18],
    double F[18][22],
    double C[18][24],
    double D[24][22],
    double G[16][22]) {

    int i, j, k;

    // First matrix multiplication: E = A * B
    for (i = 0; i < 16; i++) {
        for (j = 0; j < 18; j++) {
            E[i][j] = 0.0;
            for (k = 0; k < 20; k++) {
                E[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    // Second matrix multiplication: F = C * D
    for (i = 0; i < 18; i++) {
        for (j = 0; j < 22; j++) {
            F[i][j] = 0.0;
            for (k = 0; k < 24; k++) {
                F[i][j] += C[i][k] * D[k][j];
            }
        }
    }

    // Third matrix multiplication: G = E * F
    for (i = 0; i < 16; i++) {
        for (j = 0; j < 22; j++) {
            G[i][j] = 0.0;
            for (k = 0; k < 18; k++) {
                G[i][j] += E[i][k] * F[k][j];
            }
        }
    }
}
