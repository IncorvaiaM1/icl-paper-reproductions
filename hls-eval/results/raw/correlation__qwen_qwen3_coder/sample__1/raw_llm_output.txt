#include "correlation.h"

void kernel_correlation(
    double float_n,
    double data[32][28],
    double corr[28][28],
    double mean[28],
    double stddev[28]) {

    int i, j, k;

    // Compute mean for each attribute
    for (j = 0; j < 28; j++) {
        mean[j] = 0.0;
        for (i = 0; i < 32; i++) {
            mean[j] += data[i][j];
        }
        mean[j] /= float_n;
    }

    // Compute standard deviation for each attribute
    for (j = 0; j < 28; j++) {
        stddev[j] = 0.0;
        for (i = 0; i < 32; i++) {
            double diff = data[i][j] - mean[j];
            stddev[j] += diff * diff;
        }
        stddev[j] /= float_n;
        stddev[j] = sqrt(stddev[j]);
    }

    // Normalize the data in-place
    for (i = 0; i < 32; i++) {
        for (j = 0; j < 28; j++) {
            if (stddev[j] != 0) {
                data[i][j] = (data[i][j] - mean[j]) / stddev[j];
            } else {
                data[i][j] = 0.0;
            }
        }
    }

    // Compute correlation matrix
    for (i = 0; i < 28; i++) {
        for (j = 0; j < 28; j++) {
            if (i == j) {
                corr[i][j] = 1.0;
            } else {
                double sum = 0.0;
                for (k = 0; k < 32; k++) {
                    sum += data[k][i] * data[k][j];
                }
                corr[i][j] = sum / float_n;
            }
        }
    }
}
