#include "durbin.h"

void kernel_durbin(double r[40], double y[40]) {
    double alpha = -r[0];
    double beta = 1.0;
    double gamma[40];
    double s[40];

    y[0] = alpha;

    for (int k = 1; k < 40; k++) {
        beta = (1.0 - alpha * alpha) * beta;
        alpha = -r[k];

        for (int i = 0; i < k; i++) {
            alpha -= r[k - i - 1] * y[i];
        }

        alpha = alpha / beta;

        for (int i = 0; i < k; i++) {
            gamma[i] = y[i];
        }

        y[k] = alpha;

        for (int i = 0; i < k; i++) {
            y[i] = gamma[i] - alpha * gamma[k - i - 1];
        }
    }
}
