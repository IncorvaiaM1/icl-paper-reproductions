#include "durbin.h"

void kernel_durbin(double r[40], double y[40]) {
    double alpha = r[0];
    double beta = -r[0];
    double z[40];
    double y_old[40];

    y[0] = -r[0];
    z[0] = 1.0;

    for (int k = 1; k < 40; k++) {
        // Save previous y values
        for (int i = 0; i < k; i++) {
            y_old[i] = y[i];
        }

        beta = 0.0;
        for (int i = 0; i < k; i++) {
            beta += r[k - i - 1] * y_old[i];
        }
        beta = -r[k] - beta;

        alpha = alpha - (beta * beta) / alpha;

        // Update y
        for (int i = 0; i < k; i++) {
            y[i] = y_old[i] + (beta / alpha) * z[i];
        }
        y[k] = beta / alpha;

        // Update z
        for (int i = 0; i < k; i++) {
            z[k - 1 - i] = z[i];
        }
        z[k] = 1.0;
        for (int i = 0; i < k; i++) {
            z[i] = z[i] - (beta / alpha) * z[k - 1 - i];
        }
    }
}
