```c
#include "sort_merge.h"

void merge(TYPE a[SIZE], int start, int m, int stop) {
    TYPE temp[SIZE];
    int i, j, k;
    
    // Copy left half to temp array
    for (i = start; i <= m; i++) {
        temp[i] = a[i];
    }
    
    // Copy right half to temp array in reverse order
    for (j = m+1; j <= stop; j++) {
        temp[j] = a[stop - (j - (m+1))];
    }
    
    // Merge the two halves back into original array
    i = start;
    j = stop;
    for (k = start; k <= stop; k++) {
        if (temp[i] <= temp[j]) {
            a[k] = temp[i++];
        } else {
            a[k] = temp[j--];
        }
    }
}

void ms_mergesort(TYPE a[SIZE]) {
    int width;
    int i;
    
    // Bottom-up merge sort
    for (width = 1; width < SIZE; width = 2 * width) {
        for (i = 0; i < SIZE; i = i + 2 * width) {
            int start = i;
            int mid = i + width - 1;
            int end = i + 2 * width - 1;
            
            // Don't exceed array bounds
            if (mid >= SIZE - 1) {
                mid = SIZE - 1;
            }
            if (end >= SIZE - 1) {
                end = SIZE - 1;
            }
            
            merge(a, start, mid, end);
        }
    }
}
```

### Key Points:

1. **Merge Sort Implementation**:
   - Uses a bottom-up (iterative) approach instead of the traditional recursive approach
   - Divides the array into subarrays of increasing width (1, 2, 4, 8, etc.)
   - Merges adjacent subarrays at each step

2. **Merge Function**:
   - Takes start, middle, and end indices to define the subarrays to merge
   - Uses a temporary array to store and merge the subarrays
   - Merges by comparing elements from both subarrays and placing them in order

3. **Performance**:

   - Space complexity: O(n) for the temporary array
   - Stable sort (maintains order of equal elements)
