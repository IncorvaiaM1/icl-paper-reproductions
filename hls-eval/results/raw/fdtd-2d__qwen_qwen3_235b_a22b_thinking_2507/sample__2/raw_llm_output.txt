#include "fdtd-2d.h"

void kernel_fdtd_2d(double ex[20][30], double ey[20][30], double hz[20][30], double _fict_[20]) {
    const double C_hzh = 1.0;
    const double C_hze = 0.7;
    const double C_exe = 1.0;
    const double C_exh = 0.5;
    const double C_eye = 1.0;
    const double C_eyh = -0.5;

    for (int t = 0; t < 20; t++) {
        // Update Hz field (using previous time step's Ex and Ey)
        for (int i = 0; i < 19; i++) {
            for (int j = 0; j < 29; j++) {
                hz[i][j] = C_hzh * hz[i][j] + 
                           C_hze * (ex[i][j+1] - ex[i][j] - ey[i+1][j] + ey[i][j]);
            }
        }

        // Update Ex field (using current time step's Hz)
        for (int i = 0; i < 20; i++) {
            for (int j = 1; j < 30; j++) {
                ex[i][j] = C_exe * ex[i][j] + 
                           C_exh * (hz[i][j] - hz[i][j-1]);
            }
        }

        // Update Ey field (using current time step's Hz)
        for (int i = 1; i < 20; i++) {
            for (int j = 0; j < 30; j++) {
                ey[i][j] = C_eye * ey[i][j] + 
                           C_eyh * (hz[i][j] - hz[i-1][j]);
            }
        }

        // Apply fictitious source at boundary
        for (int j = 0; j < 30; j++) {
            ex[0][j] = _fict_[t];
        }
    }
}
