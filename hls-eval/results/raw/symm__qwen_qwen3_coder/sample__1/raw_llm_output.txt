#include "symm.h"

void kernel_symm(
    double alpha,
    double beta,
    double C[20][30],
    double A[20][20],
    double B[20][30]) {

    int i, j, k;
    double temp;

    // Iterate over rows of C
    for (i = 0; i < 20; i++) {
        // Iterate over columns of C
        for (j = 0; j < 30; j++) {
            temp = 0.0;
            // Compute alpha * A * B
            for (k = 0; k < 20; k++) {
                // Since A is symmetric and stored as lower triangular,
                // we use A[i][k] if i >= k, otherwise A[k][i]
                if (i >= k) {
                    temp += alpha * A[i][k] * B[k][j];
                } else {
                    temp += alpha * A[k][i] * B[k][j];
                }
            }
            // Compute beta * C and add to result
            C[i][j] = temp + beta * C[i][j];
        }
    }
}
