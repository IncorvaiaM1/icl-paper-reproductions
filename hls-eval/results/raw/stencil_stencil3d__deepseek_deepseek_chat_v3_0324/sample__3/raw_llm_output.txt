#include "stencil_stencil3d.h"

void stencil3d(TYPE C[2], TYPE orig[SIZE], TYPE sol[SIZE]) {
    // Handle boundary conditions by copying original values
    for (int i = 0; i < row_size; i++) {
        for (int j = 0; j < col_size; j++) {
            for (int k = 0; k < height_size; k++) {
                if (i == 0 || i == row_size - 1 || 
                    j == 0 || j == col_size - 1 || 
                    k == 0 || k == height_size - 1) {
                    int idx = INDX(row_size, col_size, i, j, k);
                    sol[idx] = orig[idx];
                }
            }
        }
    }

    // Apply stencil computation to interior points
    for (int i = 1; i < row_size - 1; i++) {
        for (int j = 1; j < col_size - 1; j++) {
            for (int k = 1; k < height_size - 1; k++) {
                int idx = INDX(row_size, col_size, i, j, k);
                int idx_kp1 = INDX(row_size, col_size, i, j, k + 1);
                int idx_km1 = INDX(row_size, col_size, i, j, k - 1);
                int idx_jp1 = INDX(row_size, col_size, i, j + 1, k);
                int idx_jm1 = INDX(row_size, col_size, i, j - 1, k);
                int idx_ip1 = INDX(row_size, col_size, i + 1, j, k);
                int idx_im1 = INDX(row_size, col_size, i - 1, j, k);

                TYPE sum = orig[idx_kp1] + orig[idx_km1] + 
                          orig[idx_jp1] + orig[idx_jm1] + 
                          orig[idx_ip1] + orig[idx_im1];

                sol[idx] = C[0] * orig[idx] + C[1] * sum;
            }
        }
    }
}
