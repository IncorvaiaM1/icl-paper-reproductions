#include "mvt.h"

void kernel_mvt(double x1[40], double x2[40], double y_1[40], double y_2[40], double A[40][40]) {
    for (int i = 0; i < 40; i++) {
        for (int j = 0; j < 40; j++) {
            x1[i] += A[i][j] * y_1[j];
        }
    }

    for (int i = 0; i < 40; i++) {
        for (int j = 0; j < 40; j++) {
            x2[i] += A[j][i] * y_2[j];
        }
    }
}
