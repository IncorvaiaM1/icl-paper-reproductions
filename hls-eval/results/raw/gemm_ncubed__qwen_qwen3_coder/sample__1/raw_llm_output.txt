#include "gemm_ncubed.h"

void gemm(TYPE m1[N], TYPE m2[N], TYPE prod[N]) {
    // Iterate over rows of m1
    for (int i = 0; i < row_size; i++) {
        // Iterate over columns of m2
        for (int j = 0; j < col_size; j++) {
            // Initialize accumulator for dot product
            TYPE sum = 0.0;
            // Perform dot product of row i of m1 with column j of m2
            for (int k = 0; k < col_size; k++) {
                sum += m1[i * col_size + k] * m2[k * col_size + j];
            }
            // Store result in prod matrix
            prod[i * col_size + j] = sum;
        }
    }
}
