#include "overlapping.h"

void Overlapping(double *result, int epsilon[N]) {
    // Pattern to match: 9 consecutive ones
    const int pattern_length = 9;
    char sequence[pattern_length];
    for (int i = 0; i < pattern_length; i++) {
        sequence[i] = 1;
    }

    // Initialize nu array to zero
    unsigned int nu[6] = {0};

    // Predefined weights for chi-square computation
    double pi[6] = {
        0.05775804494884893,
        0.14032316536906422,
        0.17251373125231415,
        0.18527752645552832,
        0.1926504296293746,
        0.2514771023448697
    };

    // Sliding window to count pattern matches
    const int window_size = 1024; // As per standard for this test
    const int num_windows = N - window_size + 1;

    // Process each window
    for (int window_start = 0; window_start < num_windows; window_start++) {
        int match_count = 0;

        // Check for pattern matches within the current window
        for (int i = 0; i <= window_size - pattern_length; i++) {
            int is_match = 1;
            for (int j = 0; j < pattern_length; j++) {
                if (epsilon[window_start + i + j] != sequence[j]) {
                    is_match = 0;
                    break;
                }
            }
            if (is_match) {
                match_count++;
            }
        }

        // Update nu based on match count
        if (match_count >= 5) {
            nu[5]++;
        } else {
            nu[match_count]++;
        }
    }

    // Compute chi-square statistic
    double chi_square = 0.0;
    double expected;
    for (int i = 0; i < 6; i++) {
        expected = pi[i] * num_windows;
        if (expected > 0) {
            double diff = (double)nu[i] - expected;
            chi_square += (diff * diff) / expected;
        }
    }

    *result = chi_square;
}
