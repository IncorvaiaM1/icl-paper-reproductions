#include "cholesky.h"
#include <cmath>

void kernel_cholesky(double A[40][40]) {
    double sum;
    for (int i = 0; i < 40; i++) {
        for (int j = 0; j <= i; j++) {
            sum = A[i][j];
            if (i == j) {
                // Diagonal elements
                for (int k = 0; k < j; k++) {
                    sum -= A[j][k] * A[j][k];
                }
                A[j][j] = sqrt(sum);
            } else {
                // Non-diagonal elements
                for (int k = 0; k < j; k++) {
                    sum -= A[i][k] * A[j][k];
                }
                A[i][j] = sum / A[j][j];
            }
        }
    }
    
    // Zero out the upper triangular part
    for (int i = 0; i < 40; i++) {
        for (int j = i+1; j < 40; j++) {
            A[i][j] = 0.0;
        }
    }
}
