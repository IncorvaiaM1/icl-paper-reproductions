#include "jacobi-1d.h"

void kernel_jacobi_1d(double A[30], double B[30]) {
    const int n = 30;
    const int tsteps = 20;

    for (int t = 0; t < tsteps; ++t) {
        for (int i = 1; i < n - 1; ++i) {
            B[i] = (A[i - 1] + A[i] + A[i + 1]) / 3.0;
        }
        for (int i = 1; i < n - 1; ++i) {
            A[i] = B[i];
        }
    }
}
