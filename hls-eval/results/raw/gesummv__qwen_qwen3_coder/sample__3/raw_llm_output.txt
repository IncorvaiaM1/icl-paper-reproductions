#include "gesummv.h"

void kernel_gesummv(
    double alpha,
    double beta,
    double A[30][30],
    double B[30][30],
    double tmp[30],
    double x[30],
    double y[30]) {

    int i, j;

    loop_i:
    for (i = 0; i < 30; i++) {
        double sum_Ax = 0.0;
        double sum_Bx = 0.0;

        loop_j:
        for (j = 0; j < 30; j++) {
            sum_Ax += A[i][j] * x[j];
            sum_Bx += B[i][j] * x[j];
        }

        tmp[i] = alpha * sum_Ax + beta * sum_Bx;
        y[i] = tmp[i];
    }
}
