void IP(des_state_t *state, des_block_t *in) {
    unsigned char temp[8];
    memcpy(temp, in, 8);

    unsigned int left = 0, right = 0;

    // Apply IP permutation
    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_TABLE[i] - 1;
        int byte_index = bit_pos / 8;
        int bit_index = 7 - (bit_pos % 8);
        unsigned int bit = (temp[byte_index] >> bit_index) & 0x01;
        left |= (bit << (31 - i));
    }

    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_TABLE[i + 32] - 1;
        int byte_index = bit_pos / 8;
        int bit_index = 7 - (bit_pos % 8);
        unsigned int bit = (temp[byte_index] >> bit_index) & 0x01;
        right |= (bit << (31 - i));
    }

    (*state)[0] = left;
    (*state)[1] = right;
}

void InvIP(des_state_t *state, des_block_t *out) {
    unsigned char temp[8] = {0};

    // Apply inverse IP permutation
    for (int i = 0; i < 64; i++) {
        int perm_bit = INV_IP_TABLE[i] - 1;
        int byte_index = perm_bit / 8;
        int bit_index = 7 - (perm_bit % 8);

        unsigned int src_byte, src_bit;
        if (i < 32) {
            src_byte = 0;
            src_bit = i;
        } else {
            src_byte = 1;
            src_bit = i - 32;
        }

        unsigned int bit_val = ((*state)[src_byte] >> (31 - src_bit)) & 0x01;
        temp[byte_index] |= (bit_val << bit_index);
    }

    memcpy(out, temp, 8);
}

unsigned int f(unsigned int right, des_subkey_t *subkey) {
    unsigned char expanded[6] = {0}; // 48 bits
    unsigned char sbox_output[4] = {0}; // 32 bits
    unsigned int result = 0;

    // Expansion permutation
    for (int i = 0; i < 48; i++) {
        int bit_pos = E_TABLE[i] - 1;
        unsigned int bit = (right >> (31 - bit_pos)) & 0x01;
        int byte_index = i / 8;
        int bit_index = 7 - (i % 8);
        expanded[byte_index] |= (bit << bit_index);
    }

    // XOR with subkey
    for (int i = 0; i < 6; i++) {
        expanded[i] ^= (*subkey)[i];
    }

    // S-box substitution
    for (int sbox_index = 0; sbox_index < 8; sbox_index++) {
        unsigned char six_bits = 0;
        int start_bit = sbox_index * 6;

        for (int i = 0; i < 6; i++) {

            int bit_index = 7 - ((start_bit + i) % 8);
            unsigned int bit = (expanded[byte_index] >> bit_index) & 0x01;
            six_bits |= (bit << (5 - i));
        }

        // S-box row: bits 0 and 5
        unsigned char row = ((six_bits >> 5) & 0x01) | ((six_bits & 0x01) << 1);
        // S-box column: bits 1-4
        unsigned char col = (six_bits >> 1) & 0x0F;
        unsigned char sbox_val;

        switch (sbox_index) {
            case 0: sbox_val = sbox1[row * 16 + col]; break;

            case 2: sbox_val = sbox3[row * 16 + col]; break;
            case 3: sbox_val = sbox4[row * 16 + col]; break;

            case 5: sbox_val = sbox6[row * 16 + col]; break;
            case 6: sbox_val = sbox7[row * 16 + col]; break;
            case 7: sbox_val = sbox8[row * 16 + col]; break;
            default: sbox_val = 0;
        }

        // Place 4-bit result in sbox_output
        int out_byte = sbox_index / 2;
        int out_bit = (sbox_index % 2) * 4;
        sbox_output[out_byte] |= (sbox_val << (4 - out_bit));
    }

    // P-box permutation
    for (int i = 0; i < 32; i++) {
        int bit_pos = P_TABLE[i] - 1;
        int byte_index = bit_pos / 8;
        int bit_index = 7 - (bit_pos % 8);
        unsigned int bit = (sbox_output[byte_index] >> bit_index) & 0x01;
        result |= (bit << (31 - i));
    }

    return result;
}

void des_key_setup(const unsigned char key[], des_key_t schedule, DES_MODE mode) {
    unsigned char pc1_key[7] = {0}; // 56 bits after PC-1
    unsigned char c[17][4] = {0};   // 28 bits each
    unsigned char d[17][4] = {0};   // 28 bits each

    // Apply PC-1 permutation to get 56-bit key
    for (int i = 0; i < 56; i++) {
        int bit_pos = PC1_TABLE[i] - 1;
        int src_byte = bit_pos / 8;
        int src_bit = 7 - (bit_pos % 8);
        unsigned int bit = (key[src_byte] >> src_bit) & 0x01;
        int dest_byte = i / 8;
        int dest_bit = 7 - (i % 8);
        pc1_key[dest_byte] |= (bit << dest_bit);
    }

    // Split into C and D (28 bits each)
    for (int i = 0; i < 28; i++) {
        int byte_index = i / 8;
        int bit_index = 7 - (i % 8);
        unsigned int bit = (pc1_key[byte_index] >> bit_index) & 0x01;
        int dest_byte = i / 8;
        int dest_bit = 7 - (i % 8);
        c[0][dest_byte] |= (bit << dest_bit);
    }

    for (int i = 0; i < 28; i++) {
        int bit_pos = i + 28;
        int byte_index = bit_pos / 8;
        int bit_index = 7 - (bit_pos % 8);
        unsigned int bit = (pc1_key[byte_index] >> bit_index) & 0x01;
        int dest_byte = i / 8;
        int dest_bit = 7 - (i % 8);
        d[0][dest_byte] |= (bit << dest_bit);
    }

    // Perform left shifts and generate subkeys
    for (int round = 1; round <= 16; round++) {
        int shifts = LEFT_SHIFTS[round - 1];

        // Left shift C
        unsigned int c_combined = (c[round-1][0] << 20) | (c[round-1][1] << 12) | (c[round-1][2] << 4) | (c[round-1][3] >> 4);
        c_combined = ((c_combined << shifts) | (c_combined >> (28 - shifts))) & 0x0FFFFFFF;
        c[round][0] = (c_combined >> 20) & 0xFF;
        c[round][1] = (c_combined >> 12) & 0xFF;
        c[round][2] = (c_combined >> 4) & 0xFF;
        c[round][3] = (c_combined & 0x0F) << 4;

        // Left shift D
        unsigned int d_combined = (d[round-1][0] << 20) | (d[round-1][1] << 12) | (d[round-1][2] << 4) | (d[round-1][3] >> 4);
        d_combined = ((d_combined << shifts) | (d_combined >> (28 - shifts))) & 0x0FFFFFFF;
        d[round][0] = (d_combined >> 20) & 0xFF;
        d[round][1] = (d_combined >> 12) & 0xFF;
        d[round][2] = (d_combined >> 4) & 0xFF;
        d[round][3] = (d_combined & 0x0F) << 4;

        // Apply PC-2 to get 48-bit subkey
        unsigned char cd[7]; // 56 bits combined
        cd[0] = c[round][0];
        cd[1] = c[round][1];
        cd[2] = c[round][2];
        cd[3] = (c[round][3] & 0xF0) | ((d[round][0] & 0xF0) >> 4);
        cd[4] = ((d[round][0] & 0x0F) << 4) | ((d[round][1] & 0xF0) >> 4);
        cd[5] = ((d[round][1] & 0x0F) << 4) | ((d[round][2] & 0xF0) >> 4);
        cd[6] = ((d[round][2] & 0x0F) << 4) | ((d[round][3] & 0xF0) >> 4);

        for (int i = 0; i < 48; i++) {
            int bit_pos = PC2_TABLE[i] - 1;
            int src_byte = bit_pos / 8;
            int src_bit = 7 - (bit_pos % 8);
            unsigned int bit = (cd[src_byte] >> src_bit) & 0x01;
            int dest_byte = i / 8;
            int dest_bit = 7 - (i % 8);
            schedule[round-1][dest_byte] |= (bit << dest_bit);
        }
    }

    // For decryption, reverse the order of subkeys
    if (mode == DES_DECRYPT) {
        des_subkey_t temp_key;
        for (int i = 0; i < 8; i++) {
            memcpy(temp_key, schedule[i], 6);
            memcpy(schedule[i], schedule[15 - i], 6);
            memcpy(schedule[15 - i], temp_key, 6);
        }
    }
}

void des_crypt(des_block_t *in, des_block_t *out, des_key_t *key) {
    des_state_t state;
    
    // Initial permutation
    IP(&state, in);

    // 16 rounds of Feistel network
    for (int round = 0; round < 16; round++) {
        unsigned int temp = state[0];
        state[0] = state[1];
        state[1] = temp ^ f(state[1], &(*key)[round]);
    }

    // Swap left and right
    unsigned int temp = state[0];
    state[0] = state[1];
    state[1] = temp;

    // Final permutation
    InvIP(&state, out);
}
