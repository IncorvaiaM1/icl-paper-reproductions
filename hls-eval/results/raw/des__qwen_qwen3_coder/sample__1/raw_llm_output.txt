void IP(des_state_t *state, des_block_t *in) {
    unsigned int L = 0, R = 0;
    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_table[i] - 1;
        L |= BITNUM(in, bit_pos, 31 - i);
    }
    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_table[i + 32] - 1;
        R |= BITNUM(in, bit_pos, 31 - i);
    }
    state[0] = L;
    state[1] = R;
}

void InvIP(des_state_t *state, des_block_t *out) {
    unsigned int temp[2] = {state[0], state[1]};
    unsigned int L = 0, R = 0;
    for (int i = 0; i < 64; i++) {
        int bit_pos = InvIP_table[i] - 1;
        unsigned int val;
        if (bit_pos < 32)
            val = BITNUMINTR(temp[0], bit_pos, 7 - (i % 8));
        else
            val = BITNUMINTR(temp[1], bit_pos - 32, 7 - (i % 8));

        out[i / 8] |= val;
    }
}

unsigned int f(unsigned int state, des_subkey_t *key) {
    unsigned int expanded = 0;
    for (int i = 0; i < 48; i++) {
        int bit_pos = E_table[i] - 1;
        expanded |= BITNUMINTR(state, bit_pos, 47 - i);
    }

    unsigned int xor_result = 0;
    for (int i = 0; i < 6; i++) {
        unsigned int key_byte = (*key)[i];
        unsigned int exp_chunk = (expanded >> (42 - 8 * i)) & 0xFF;
        if (i == 5) {
            exp_chunk = (expanded >> (42 - 8 * i)) & 0x03;
            key_byte &= 0x03;
        }
        xor_result |= ((exp_chunk ^ key_byte) << (40 - 8 * i));
    }

    unsigned int sbox_output = 0;
    for (int i = 0; i < 8; i++) {
        unsigned int sbox_input = (xor_result >> (42 - 6 * i)) & 0x3F;
        unsigned int row = ((sbox_input & 0x20) >> 4) | (sbox_input & 0x01);
        unsigned int col = (sbox_input & 0x1E) >> 1;
        unsigned int sbox_index = row * 16 + col;
        unsigned int sbox_val;
        switch (i) {
            case 0: sbox_val = sbox1[sbox_index]; break;
            case 1: sbox_val = sbox2[sbox_index]; break;
            case 2: sbox_val = sbox3[sbox_index]; break;
            case 3: sbox_val = sbox4[sbox_index]; break;
            case 4: sbox_val = sbox5[sbox_index]; break;
            case 5: sbox_val = sbox6[sbox_index]; break;
            case 6: sbox_val = sbox7[sbox_index]; break;
            case 7: sbox_val = sbox8[sbox_index]; break;
        }
        sbox_output |= (sbox_val << (28 - 4 * i));
    }

    unsigned int p_result = 0;
    for (int i = 0; i < 32; i++) {
        int bit_pos = P_table[i] - 1;
        p_result |= BITNUMINTR(sbox_output, bit_pos, 31 - i);
    }

    return p_result;
}

void des_key_setup(const unsigned char key[], des_key_t schedule, DES_MODE mode) {
    unsigned char PC1_key[7];
    for (int i = 0; i < 56; i++) {
        int bit_pos = PC1_table[i] - 1;
        PC1_key[i / 8] |= BITNUM(key, bit_pos, 7 - (i % 8));
    }

    unsigned int C[17], D[17];
    C[0] = ((unsigned int)PC1_key[0] << 24) | ((unsigned int)PC1_key[1] << 16) |
           ((unsigned int)PC1_key[2] << 8) | PC1_key[3];
    D[0] = ((unsigned int)PC1_key[3] << 28) | ((unsigned int)PC1_key[4] << 20) |
           ((unsigned int)PC1_key[5] << 12) | ((unsigned int)PC1_key[6] << 4);

    for (int i = 1; i <= 16; i++) {
        int shift = shifts[i - 1];
        C[i] = (C[i - 1] << shift) | (C[i - 1] >> (28 - shift));
        D[i] = (D[i - 1] << shift) | (D[i - 1] >> (28 - shift));
        C[i] &= 0xFFFFFFF0;
        D[i] &= 0xFFFFFFF0;
    }

    for (int round = 0; round < 16; round++) {
        int idx = (mode == DES_ENCRYPT) ? round : (15 - round);
        unsigned int combined = (C[idx + 1] >> 4) | (D[idx + 1] << 28);
        for (int i = 0; i < 48; i++) {
            int bit_pos = PC2_table[i] - 1;
            schedule[round][i / 8] |= BITNUMINTR(combined, bit_pos, 7 - (i % 8));
        }
    }
}

void des_crypt(des_block_t *in, des_block_t *out, des_key_t *key) {
    des_state_t state;
    IP(state, in);

    for (int round = 0; round < 16; round++) {
        unsigned int temp = state[0];
        state[0] = state[1];
        state[1] = temp ^ f(state[1], &(*key)[round]);
    }

    unsigned int temp = state[0];
    state[0] = state[1];
    state[1] = temp;

    InvIP(state, out);
}
