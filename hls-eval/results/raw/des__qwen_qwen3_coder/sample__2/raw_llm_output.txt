void IP(des_state_t *state, des_block_t *in) {
    unsigned int l = 0, r = 0;
    
    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_TABLE[i] - 1;
        int byte_idx = bit_pos / 8;
        int bit_idx = 7 - (bit_pos % 8);
        l |= (((in[byte_idx] >> bit_idx) & 1) << (31 - i));
    }
    
    for (int i = 0; i < 32; i++) {
        int bit_pos = IP_TABLE[i + 32] - 1;
        int byte_idx = bit_pos / 8;
        int bit_idx = 7 - (bit_pos % 8);
        r |= (((in[byte_idx] >> bit_idx) & 1) << (31 - i));
    }
    
    (*state)[0] = l;
    (*state)[1] = r;
}

void des_key_setup(const unsigned char key[], des_key_t schedule, DES_MODE mode) {
    unsigned char pc1_key[7] = {0};
    unsigned int c, d;
    
    // PC-1 permutation
    for (int i = 0; i < 56; i++) {
        int bit_pos = PC1_TABLE[i] - 1;
        int src_byte = bit_pos / 8;
        int src_bit = 7 - (bit_pos % 8);
        int dest_byte = i / 8;
        int dest_bit = 7 - (i % 8);
        
        if ((key[src_byte] >> src_bit) & 1) {
            pc1_key[dest_byte] |= (1 << dest_bit);
        }
    }
    
    // Split into C and D registers
    c = ((unsigned int)pc1_key[0] << 21) |
        ((unsigned int)pc1_key[1] << 13) |
        ((unsigned int)pc1_key[2] << 5) |
        ((unsigned int)pc1_key[3] >> 3);
        
    d = ((unsigned int)(pc1_key[3] & 0x07) << 25) |
        ((unsigned int)pc1_key[4] << 17) |
        ((unsigned int)pc1_key[5] << 9) |
        ((unsigned int)pc1_key[6] << 1);
    
    // Generate 16 subkeys
    for (int round = 0; round < 16; round++) {
        // Left shift C and D
        c = ((c << LEFT_SHIFTS[round]) | (c >> (28 - LEFT_SHIFTS[round]))) & 0x0FFFFFFF;
        d = ((d << LEFT_SHIFTS[round]) | (d >> (28 - LEFT_SHIFTS[round]))) & 0x0FFFFFFF;
        
        // PC-2 permutation to create subkey
        unsigned int cd = (c << 28) | d;
        for (int i = 0; i < 48; i++) {
            int bit_pos = PC2_TABLE[i] - 1;
            int src_bit = 55 - bit_pos;
            int dest_byte = i / 8;
            int dest_bit = 7 - (i % 8);
            
            if ((cd >> src_bit) & 1) {
                schedule[round][dest_byte] |= (1 << dest_bit);
            }
        }
    }
    
    // For decryption, reverse the order of subkeys
    if (mode == DES_DECRYPT) {
        des_key_t temp_schedule;
        for (int i = 0; i < 16; i++) {
            for (int j = 0; j < 6; j++) {
                temp_schedule[i][j] = schedule[15 - i][j];
            }
        }
        for (int i = 0; i < 16; i++) {
            for (int j = 0; j < 6; j++) {
                schedule[i][j] = temp_schedule[i][j];
            }
        }
    }
}

void des_crypt(des_block_t *in, des_block_t *out, des_key_t *key) {
    des_state_t state;
    
    // Initial permutation
    IP(&state, in);
    
    // 16 rounds of Feistel network
    for (int round = 0; round < 16; round++) {
        unsigned int temp = state[0];
        state[0] = state[1];
        state[1] = temp ^ f(state[1], &(*key)[round]);
    }
    
    // Swap left and right halves for final permutation
    unsigned int temp = state[0];
    state[0] = state[1];
    state[1] = temp;
    
    // Inverse initial permutation
    InvIP(&state, out);
}
