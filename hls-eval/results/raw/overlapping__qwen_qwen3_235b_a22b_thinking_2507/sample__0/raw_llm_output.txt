#include "overlapping.h"

void Overlapping(double *result, int epsilon[N]) {
    const int M = 1032;
    const int K = N / M;
    unsigned int nu[6] = {0};
    double pi[6] = {0.032258, 0.082645, 0.153846, 0.189796, 0.189796, 0.351659};

    for (int block = 0; block < K; block++) {
        int count = 0;
        for (int j = 0; j <= M - 9; j++) {
            bool match = true;
            for (int k = 0; k < 9; k++) {
                if (epsilon[block * M + j + k] != 1) {
                    match = false;
                    break;
                }
            }
            if (match) {
                count++;
            }
        }
        if (count >= 5) {
            nu[5]++;
        } else {
            nu[count]++;
        }
    }

    double chi_square = 0.0;
    for (int i = 0; i < 6; i++) {
        double expected = K * pi[i];
        double diff = nu[i] - expected;
        chi_square += (diff * diff) / expected;
    }

    *result = chi_square;
}
