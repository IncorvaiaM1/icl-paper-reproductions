#include "mix_cols.h"

static uint8_t xtime(uint8_t x) {
    return (x << 1) ^ (((x >> 7) & 1) * 0x1b);
}

void MixColumns(state_t *state) {
    for (int j = 0; j < 4; j++) {
        uint8_t s0 = (*state)[j][0];
        uint8_t s1 = (*state)[j][1];
        uint8_t s2 = (*state)[j][2];
        uint8_t s3 = (*state)[j][3];
        
        (*state)[j][0] = xtime(s0) ^ xtime(s1) ^ s1 ^ s2 ^ s3;
        (*state)[j][1] = s0 ^ xtime(s1) ^ xtime(s2) ^ s2 ^ s3;
        (*state)[j][2] = s0 ^ s1 ^ xtime(s2) ^ xtime(s3) ^ s3;
        (*state)[j][3] = xtime(s0) ^ s0 ^ s1 ^ s2 ^ xtime(s3);
    }
}
