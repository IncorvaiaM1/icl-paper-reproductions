#include "symm.h"

void kernel_symm(
    double alpha,
    double beta,
    double C[20][30],
    double A[20][20],
    double B[20][30]) {
    const int M = 20;
    const int N = 30;

    for (int i = 0; i < M; i++) {
        for (int j = 0; j < N; j++) {
            double sum = 0.0;

            for (int k = 0; k < M; k++) {
                double aik;
                if (i >= k) {
                    aik = A[i][k];
                } else {
                    aik = A[k][i];
                }
                sum += aik * B[k][j];
            }

            C[i][j] = alpha * sum + beta * C[i][j];
        }
    }
}
