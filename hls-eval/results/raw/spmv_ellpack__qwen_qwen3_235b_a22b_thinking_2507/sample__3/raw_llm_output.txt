#include "spmv_ellpack.h"

void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {
    for (int i = 0; i < N; i++) {
    #pragma HLS PIPELINE
        TYPE sum = 0;
        int base = i * L;
        for (int j = 0; j < L; j++) {
        #pragma HLS UNROLL
            sum += nzval[base + j] * vec[cols[base + j]];
        }
        out[i] = sum;
    }
}
