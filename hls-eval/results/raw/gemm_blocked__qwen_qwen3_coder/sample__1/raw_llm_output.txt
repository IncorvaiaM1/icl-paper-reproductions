#include "gemm_blocked.h"

void bbgemm(TYPE m1[N], TYPE m2[N], TYPE prod[N]) {
    // Initialize the product matrix to zero
    int i, j;
    for (i = 0; i < N; i++) {
        prod[i] = 0.0;
    }

    // Blocked matrix multiplication
    int jj, kk;
    int k, temp_j;

    // Loop over blocks
    for (jj = 0; jj < col_size; jj += block_size) {
        for (kk = 0; kk < col_size; kk += block_size) {
            // Loop over elements within a block
            for (i = 0; i < row_size; i++) {
                for (k = kk; k < kk + block_size; k++) {
                    temp_j = i * col_size;
                    for (j = jj; j < jj + block_size; j++) {
                        prod[temp_j + j] += m1[temp_j + k] * m2[k * col_size + j];
                    }
                }
            }
        }
    }
}
