#include "correlation.h"

void kernel_correlation(
    double float_n,
    double data[32][28],
    double corr[28][28],
    double mean[28],
    double stddev[28]) {

    int i, j, k;

    // Compute mean of each attribute (column)
    for (j = 0; j &lt; 28; j++) {
        double sum = 0.0;
        for (i = 0; i &lt; 32; i++) {
            sum += data[i][j];
        }
        mean[j] = sum / float_n;
    }

    // Compute standard deviation of each attribute
    for (j = 0; j &lt; 28; j++) {
        double sq_sum = 0.0;
        for (i = 0; i &lt; 32; i++) {
            double diff = data[i][j] - mean[j];
            sq_sum += diff * diff;
        }
        stddev[j] = std::sqrt(sq_sum / float_n);
    }

    // Compute correlation matrix
    for (i = 0; i &lt; 28; i++) {
        for (j = 0; j &lt; 28; j++) {
            double cov = 0.0;
            for (k = 0; k &lt; 32; k++) {
                cov += (data[k][i] - mean[i]) * (data[k][j] - mean[j]);
            }
            corr[i][j] = cov / (float_n * stddev[i] * stddev[j]);
        }
    }
}
