#include "sort_merge.h"

void merge(TYPE a[SIZE], int start, int m, int stop) {
    TYPE temp[SIZE];
    int i, j, k;

    merge_label1:
    for (i = start; i < m; i++) {
        temp[i] = a[i];
    }

    merge_label2:
    for (j = m; j < stop; j++) {
        temp[j] = a[j];
    }

    i = start;
    j = m;

    merge_label3:
    for (k = start; k < stop; k++) {
        if (i < m && (j >= stop || temp[i] <= temp[j])) {
            a[k] = temp[i];
            i++;
        } else {
            a[k] = temp[j];
            j++;
        }
    }
}

void ms_mergesort(TYPE a[SIZE]) {
    int width, start;

    for (width = 1; width < SIZE; width = width * 2) {
        for (start = 0; start < SIZE; start = start + 2 * width) {
            int m = start + width;
            int stop = start + 2 * width;

            if (m > SIZE) {
                m = SIZE;
            }
            if (stop > SIZE) {
                stop = SIZE;
            }

            merge(a, start, m, stop);
        }
    }
}
