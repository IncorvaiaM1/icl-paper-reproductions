static void merge(TYPE a[SIZE], int start, int m, int stop) {
    TYPE temp[SIZE];
    int n1 = m - start;

    copy_left:

    }

    copy_right:
    for (int j = 0; j &lt; n2; j++) {
        temp[n1 + j] = a[m + j];
    }

    int i = 0;

    merge_label3:

}

void ms_mergesort(TYPE a[SIZE]) {
    for (int width = 1; width &lt; SIZE; width = width * 2) {
        for (int start = 0; start &lt; SIZE; start = start + 2 * width) {
            int mid = start + width;
            int stop = start + 2 * width;

            if (mid &gt; SIZE) mid = SIZE;
            if (stop &gt; SIZE) stop = SIZE;

            if (mid &lt; stop) {
                merge(a, start, mid, stop);

        }
