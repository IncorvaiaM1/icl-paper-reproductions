#include "durbin.h"

void kernel_durbin(double r[40], double y[40]) {
    // Initialize the first element of y
    y[0] = -r[0];
    // Initialize alpha for the first step
    double alpha = 1.0 - r[0] * r[0];
    
    // Iterate for the remaining elements (from index 1 to 39)
    for (int k = 1; k < 40; k++) {
        double sum = 0.0;
        // Compute the sum term for the current step
        for (int j = 0; j < k; j++) {
            sum += y[j] * r[k - 1 - j];
        }
        // Calculate gamma using the current sum and alpha
        double gamma = (r[k] + sum) / alpha;
        // Update the previous y values
        for (int j = 0; j < k; j++) {
            y[j] = y[j] - gamma * y[k - 1 - j];
        }
        // Set the current y value
        y[k] = -gamma;
        // Update alpha for the next iteration
        alpha = alpha * (1.0 - gamma * gamma);
    }
}
