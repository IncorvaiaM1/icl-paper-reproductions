#include "correlation.h"

void kernel_correlation(double float_n, double data[32][28], double corr[28][28], double mean[28], double stddev[28]) {
    // Compute mean for each attribute
    for (int j = 0; j < 28; j++) {
        double sum = 0.0;
        for (int k = 0; k < 32; k++) {
            sum += data[k][j];
        }
        mean[j] = sum / float_n;
    }

    // Compute standard deviation for each attribute
    for (int j = 0; j < 28; j++) {
        double sum_sq = 0.0;
        for (int k = 0; k < 32; k++) {
            double diff = data[k][j] - mean[j];
            sum_sq += diff * diff;
        }
        stddev[j] = sqrt(sum_sq / float_n);
    }

    // Compute correlation matrix
    for (int i = 0; i < 28; i++) {
        for (int j = 0; j < 28; j++) {
            double sum_p = 0.0;
            for (int k = 0; k < 32; k++) {
                double norm_i = (data[k][i] - mean[i]) / stddev[i];
                double norm_j = (data[k][j] - mean[j]) / stddev[j];
                sum_p += norm_i * norm_j;
            }
            corr[i][j] = sum_p / float_n;
        }
    }
}
