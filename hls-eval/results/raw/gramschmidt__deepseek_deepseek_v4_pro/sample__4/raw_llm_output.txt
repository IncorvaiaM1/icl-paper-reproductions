#include "gramschmidt.h"

void kernel_gramschmidt(double A[20][30], double R[30][30], double Q[20][30]) {
    const int M = 20;
    const int N = 30;

    for (int k = 0; k < N; k++) {
        double norm = 0.0;

        for (int i = 0; i < M; i++) {
            norm += A[i][k] * A[i][k];
        }

        R[k][k] = sqrt(norm);

        for (int i = 0; i < M; i++) {
            Q[i][k] = A[i][k] / R[k][k];
        }

        for (int j = k + 1; j < N; j++) {
            double proj = 0.0;

            for (int i = 0; i < M; i++) {
                proj += Q[i][k] * A[i][j];
            }

            R[k][j] = proj;

            for (int i = 0; i < M; i++) {
                A[i][j] -= Q[i][k] * R[k][j];
            }
        }
    }
}
