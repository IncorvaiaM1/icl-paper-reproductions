#include "gramschmidt.h"

void kernel_gramschmidt(double A[20][30], double R[30][30], double Q[20][30]) {
    const int M = 20;
    const int N = 30;

    // Initialize output matrices
    for (int i = 0; i < M; i++) {
        for (int j = 0; j < N; j++) {
            Q[i][j] = 0.0;
        }
    }
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            R[i][j] = 0.0;
        }
    }

    // Modified Gram-Schmidt orthogonalization
    for (int k = 0; k < N; k++) {
        // Normalization: compute the norm of column k of A
        double nrm = 0.0;
        for (int i = 0; i < M; i++) {
            nrm += A[i][k] * A[i][k];
        }
        double sqrt_nrm = std::sqrt(nrm);
        R[k][k] = sqrt_nrm;

        // Orthogonalization: compute the k-th column of Q
        for (int i = 0; i < M; i++) {
            Q[i][k] = A[i][k] / sqrt_nrm;
        }

        // Update: subtract the projection of remaining columns onto q_k
        for (int j = k + 1; j < N; j++) {
            double dot = 0.0;
            for (int i = 0; i < M; i++) {
                dot += Q[i][k] * A[i][j];
            }
            R[k][j] = dot;

            for (int i = 0; i < M; i++) {
                A[i][j] -= dot * Q[i][k];
            }
        }
    }
}
