#include "gramschmidt.h"

void kernel_gramschmidt(double A[20][30], double R[30][30], double Q[20][30]) {
    for (int j = 0; j < 30; j++) {
        double norm = 0.0;
        for (int i = 0; i < 20; i++) {
            norm += A[i][j] * A[i][j];
        }
        R[j][j] = sqrt(norm);
        
        for (int i = 0; i < 20; i++) {
            Q[i][j] = A[i][j] / R[j][j];
        }
        
        for (int k = j + 1; k < 30; k++) {
            double dot = 0.0;
            for (int i = 0; i < 20; i++) {
                dot += Q[i][j] * A[i][k];
            }
            R[j][k] = dot;
            
            for (int i = 0; i < 20; i++) {
                A[i][k] = A[i][k] - dot * Q[i][j];
            }
        }
    }
}
