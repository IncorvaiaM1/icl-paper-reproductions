#include "gramschmidt.h"

void kernel_gramschmidt(double A[20][30], double R[30][30], double Q[20][30]) {
    const int M = 20;
    const int N = 30;

    for (int k = 0; k < N; k++) {
        // Compute the norm of column k
        double norm = 0.0;
        for (int i = 0; i < M; i++) {
            norm += A[i][k] * A[i][k];
        }
        R[k][k] = sqrt(norm);

        // Normalize column k to get Q
        for (int i = 0; i < M; i++) {
            Q[i][k] = A[i][k] / R[k][k];
        }

        // Update remaining columns j > k
        for (int j = k + 1; j < N; j++) {
            // Compute inner product for R
            double r = 0.0;
            for (int i = 0; i < M; i++) {
                r += Q[i][k] * A[i][j];
            }
            R[k][j] = r;

            // Update A's column j
            for (int i = 0; i < M; i++) {
                A[i][j] -= r * Q[i][k];
            }
        }
    }
}
