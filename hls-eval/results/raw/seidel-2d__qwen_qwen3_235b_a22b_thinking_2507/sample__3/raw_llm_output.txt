#include "seidel-2d.h"

void kernel_seidel_2d(double A[40][40]) {
    for (int t = 0; t < 20; t++) {
        for (int i = 1; i < 39; i++) {
            for (int j = 1; j < 39; j++) {
                A[i][j] = (A[i-1][j-1] + A[i-1][j] + A[i-1][j+1] +
                           A[i][j-1]   + A[i][j]   + A[i][j+1]   +
                           A[i+1][j-1] + A[i+1][j] + A[i+1][j+1]) / 9.0;
            }
        }
    }
}
