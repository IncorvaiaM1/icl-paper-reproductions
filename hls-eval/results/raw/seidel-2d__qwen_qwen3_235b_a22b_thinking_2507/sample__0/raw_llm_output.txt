#include "seidel-2d.h"

void kernel_seidel_2d(double A[40][40]) {
    double temp[40][40];
    const int n = 40;
    const int tsteps = 20;
    const double factor = 1.0 / 9.0;

    for (int t = 0; t < tsteps; t++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                temp[i][j] = A[i][j];
            }
        }

        for (int i = 1; i < n - 1; i++) {
            for (int j = 1; j < n - 1; j++) {
                A[i][j] = (temp[i-1][j-1] + temp[i-1][j] + temp[i-1][j+1] +
                           temp[i][j-1]   + temp[i][j]   + temp[i][j+1] +
                           temp[i+1][j-1] + temp[i+1][j] + temp[i+1][j+1]) * factor;
            }
        }
    }
}
