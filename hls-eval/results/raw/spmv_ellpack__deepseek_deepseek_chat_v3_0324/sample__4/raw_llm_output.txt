#include "spmv_ellpack.h"

void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {
    #pragma HLS INTERFACE m_axi port=nzval bundle=gmem
    #pragma HLS INTERFACE m_axi port=cols bundle=gmem
    #pragma HLS INTERFACE m_axi port=vec bundle=gmem
    #pragma HLS INTERFACE m_axi port=out bundle=gmem
    #pragma HLS INTERFACE s_axilite port=return bundle=control

    // Initialize output vector to zero
    init_out: for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE II=1
        out[i] = 0;
    }

    // Main computation loop
    row_loop: for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE II=1
        TYPE sum = 0;
        
        // Compute dot product for current row
        col_loop: for (int j = 0; j < L; j++) {
            #pragma HLS UNROLL factor=4
            int idx = i * L + j;
            int col = cols[idx];
            if (col >= 0) {  // Check for valid column index
                sum += nzval[idx] * vec[col];
            }
        }
        
        out[i] = sum;
    }
}
