   - Initialize output vector to 0
   - For each row i (0 to N-1):
     - For each non-zero element in row i (0 to L-1):
       - Multiply nzval[i*L + k] with vec[cols[i*L + k]]
       - Accumulate the result in out[i]

void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {

        out[i] = 0;

    // Compute SpMV in ELLPACK format
    for (int i = 0; i < N; i++) {

                out[i] += nzval[i * L + k] * vec[col];

        }
    }
}
