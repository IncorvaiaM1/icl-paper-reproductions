#include "doitgen.h"

void kernel_doitgen(double A[10][8][12], double C4[12][12], double sum[12]) {
    int r, q, p, s;

    for (r = 0; r < 10; r++) {
        for (q = 0; q < 8; q++) {
            // Initialize sum array to zero
            for (p = 0; p < 12; p++) {
                sum[p] = 0.0;
            }

            // Compute sum[p] = \sum_{s=0}^{11} A[r][q][s] * C4[p][s]
            for (s = 0; s < 12; s++) {
                double A_val = A[r][q][s];
                for (p = 0; p < 12; p++) {
                    sum[p] += A_val * C4[p][s];
                }
            }

            // Store result back into A[r][q][p]
            for (p = 0; p < 12; p++) {
                A[r][q][p] = sum[p];
            }
        }
    }
}
