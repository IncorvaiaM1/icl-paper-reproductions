#include "atax.h"

void kernel_atax(double A[38][42], double x[42], double y[42], double tmp[38]) {
    // First matrix-vector multiplication: tmp = A * x
    for (int i = 0; i < 38; i++) {
        tmp[i] = 0.0;
        for (int j = 0; j < 42; j++) {
            tmp[i] += A[i][j] * x[j];
        }
    }

    // Second matrix-vector multiplication: y = A^T * tmp
    for (int j = 0; j < 42; j++) {
        y[j] = 0.0;
        for (int i = 0; i < 38; i++) {
            y[j] += A[i][j] * tmp[i];
        }
    }
}
