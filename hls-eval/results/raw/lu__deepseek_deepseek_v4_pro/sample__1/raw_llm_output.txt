#include "lu.h"

void kernel_lu(double A[40][40]) {
    const int N = 40;

    for (int i = 0; i != N; ++i) {
        // Lower triangular part
        for (int j = 0; j != i; ++j) {
            for (int k = 0; k != j; ++k) {
                A[i][j] -= A[i][k] * A[k][j];
            }
            A[i][j] /= A[j][j];
        }

        // Upper triangular part including diagonal
        for (int j = i; j != N; ++j) {
            for (int k = 0; k != i; ++k) {
                A[i][j] -= A[i][k] * A[k][j];
            }
        }
    }
}
