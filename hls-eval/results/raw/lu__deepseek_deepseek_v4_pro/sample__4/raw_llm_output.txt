#include "lu.h"

void kernel_lu(double A[40][40]) {
    int i, j, k;

    for (i = 0; i < 40; i++) {
        // Lower triangular part: compute L(i,j) for j < i
        for (j = 0; j < i; j++) {
            for (k = 0; k < j; k++) {
                A[i][j] -= A[i][k] * A[k][j];
            }
            A[i][j] /= A[j][j];
        }

        // Upper triangular part: compute U(i,j) for j >= i
        for (j = i; j < 40; j++) {
            for (k = 0; k < i; k++) {
                A[i][j] -= A[i][k] * A[k][j];
            }
        }
    }
}
