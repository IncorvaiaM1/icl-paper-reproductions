#include "2mm.h"

void kernel_2mm(
    double alpha,
    double beta,
    double tmp[16][18],
    double A[16][22],
    double B[22][18],
    double C[18][24],
    double D[16][24]) {

    // Compute intermediate matrix product: tmp = A * B
    for (int i = 0; i < 16; i++) {
        for (int k = 0; k < 18; k++) {
            double sum = 0.0;
            for (int j = 0; j < 22; j++) {
                sum += A[i][j] * B[j][k];
            }
            tmp[i][k] = sum;
        }
    }

    // Compute final result: D = alpha * (tmp * C) + beta * D
    for (int i = 0; i < 16; i++) {
        for (int l = 0; l < 24; l++) {
            double sum = 0.0;
            for (int k = 0; k < 18; k++) {
                sum += tmp[i][k] * C[k][l];
            }
            D[i][l] = alpha * sum + beta * D[i][l];
        }
    }
}
