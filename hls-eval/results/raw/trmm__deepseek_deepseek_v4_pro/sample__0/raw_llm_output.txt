#include "trmm.h"

void kernel_trmm(double alpha, double A[20][20], double B[20][30]) {
    // B = alpha * A * B
    // A is lower triangular with unit diagonal, multiplication from left
    // Compute in-place from bottom row to top to avoid overwriting needed values
    for (int i = 19; i >= 0; i--) {
        for (int j = 0; j < 30; j++) {
            double sum = 0.0;
            // dot product of row i of A (up to diagonal) with column j of B
            for (int k = 0; k < i; k++) {
                sum += A[i][k] * B[k][j];
            }
            // diagonal of A is 1, so add B[i][j] and scale by alpha
            B[i][j] = alpha * (B[i][j] + sum);
        }
    }
}
