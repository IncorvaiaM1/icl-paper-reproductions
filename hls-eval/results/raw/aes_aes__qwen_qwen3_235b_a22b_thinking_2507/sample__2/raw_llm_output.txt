uint8_t rj_xtime(uint8_t x) {
    if (x & 0x80) {
        return (x << 1) ^ 0x1b;
    } else {
        return x << 1;
    }
}

uint8_t gf_alog(uint8_t x) {
    x = x % 255;
    if (x == 0) {
        return 1;
    }
    uint8_t result = 1;
    for (int i = 0; i < x; i++) {
        result = rj_xtime(result);
    }
    return result;
}

uint8_t gf_log(uint8_t x) {
    if (x == 0) {
        return 255;
    }
    uint8_t result = 0;
    uint8_t a = 1;
    while (a != x) {
        a = rj_xtime(a);
        result++;
        if (result > 255) {
            break;
        }
    }
    return result;
}

uint8_t gf_mulinv(uint8_t x) {
    if (x == 0) {
        return 0;
    }
    uint8_t logx = gf_log(x);
    return gf_alog(255 - logx);
}

uint8_t rj_sbox(uint8_t x) {
    if (x == 0) {
        return 0;
    }
    uint8_t logx = gf_log(x);
    uint8_t inv = gf_alog(255 - logx);
    return inv ^ x;
}

void aes_subBytes(uint8_t *buf) {

void aes_mixColumns(uint8_t *buf) {
    for (int col = 0; col < 4; col++) {
        uint8_t s0 = buf[col*4+0];
        uint8_t s1 = buf[col*4+1];
        uint8_t s2 = buf[col*4+2];

void aes_expandEncKey(uint8_t *k, uint8_t *rc) {
    uint8_t a0 = k[28];
    uint8_t a1 = k[29];
    uint8_t a2 = k[30];
    uint8_t a3 = k[31];

    uint8_t b3 = rj_sbox(a0);
    
    k[0] ^= b0;
    k[1] ^= b1;
    k[2] ^= b2;
    k[3] ^= b3;

void aes256_encrypt_ecb(aes256_context *ctx, uint8_t k[32], uint8_t buf[16]) {
    memcpy(ctx->key, k, 32);
    memcpy(ctx->enckey, k, 32);
    
    aes_addRoundKey(buf, ctx->enckey);
    
    for (int r = 1; r <= 14; r++) {
        if (r % 2 == 0) {
            uint8_t rc = 1 << ((r / 2) - 1);
            aes_expandEncKey(ctx->enckey, &rc);
        }
        aes_subBytes(buf);
        aes_shiftRows(buf);
        if (r < 14) {
            aes_mixColumns(buf);
        }
        aes_addRoundKey(buf, ctx->enckey + ((r % 2) * 16));
    }
