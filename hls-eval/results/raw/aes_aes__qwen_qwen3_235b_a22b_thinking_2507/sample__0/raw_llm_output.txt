uint8_t rj_xtime(uint8_t x) {
    return (x & 0x80) ? ((x << 1) ^ 0x1b) : (x << 1);
}

uint8_t gf_log(uint8_t x) {
    if (x == 0) {
        return 0;
    }
    uint8_t log = 0;
    uint8_t g = 1;
    while (g != x) {
        g = rj_xtime(g) ^ g;
        log++;
        if (log == 255) {
            break;
        }
    }
    return log;
}

uint8_t gf_alog(uint8_t x) {
    x = x % 255;
    if (x == 0) {
        return 1;
    }
    uint8_t res = 1;
    for (int i = 0; i < x; i++) {
        res = rj_xtime(res) ^ res;
    }
    return res;
}

uint8_t gf_mulinv(uint8_t x) {
    if (x == 0) {
        return 0;
    }
    uint8_t logx = gf_log(x);
    uint8_t exponent = (255 - logx) % 255;
    return gf_alog(exponent);
}

uint8_t rj_sbox(uint8_t x) {
    if (x == 0) {
        return 0;
    }
    uint8_t logx = gf_log(x);
    uint8_t inv = gf_alog(255 - logx);
    return inv ^ x;
}

void aes_mixColumns(uint8_t *buf) {

        uint8_t s0 = buf[col*4 + 0];
