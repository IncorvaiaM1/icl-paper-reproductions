#include "gemm.h"

void kernel_gemm(
    double alpha,
    double beta,
    double C[20][25],
    double A[20][30],
    double B[30][25]) {

    // Perform the GEMM operation: C = alpha * A * B + beta * C
    // Matrix dimensions:
    // A: 20 x 30
    // B: 30 x 25
    // C: 20 x 25

    // Iterate over rows of A (and C)
    for (int i = 0; i < 20; i++) {
        // Iterate over columns of B (and C)
        for (int j = 0; j < 25; j++) {
            // Compute the dot product of row i of A and column j of B
            double sum = 0.0;
            for (int k = 0; k < 30; k++) {
                sum += A[i][k] * B[k][j];
            }
            // Apply the GEMM formula: C[i][j] = alpha * sum + beta * C[i][j]
            C[i][j] = alpha * sum + beta * C[i][j];
        }
    }
}
