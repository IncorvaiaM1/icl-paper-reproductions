#include "gemm.h"

void kernel_gemm(
    double alpha,
    double beta,
    double C[20][25],
    double A[20][30],
    double B[30][25]) {

    // Temporary matrix to store the result of alpha * A * B
    double tmp[20][25];

    // Perform matrix multiplication: tmp = A * B
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 25; j++) {
            tmp[i][j] = 0.0;
            for (int k = 0; k < 30; k++) {
                tmp[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    // Perform scaling and accumulation: C = alpha * tmp + beta * C
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 25; j++) {
            C[i][j] = alpha * tmp[i][j] + beta * C[i][j];
        }
    }
}
