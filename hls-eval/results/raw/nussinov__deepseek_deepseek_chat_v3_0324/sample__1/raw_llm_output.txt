#include "nussinov.h"

void kernel_nussinov(char seq[60], int table[60][60]) {
    int n = 60;
    
    // Initialize diagonal entries
    for (int i = 0; i < n; i++) {
        table[i][i] = 0;
    }
    
    // Initialize entries just above diagonal
    for (int i = 0; i < n-1; i++) {
        int j = i + 1;
        table[i][j] = 0;
        if ((seq[i] == 'A' && seq[j] == 'T') || 
            (seq[i] == 'T' && seq[j] == 'A') || 
            (seq[i] == 'G' && seq[j] == 'C') || 
            (seq[i] == 'C' && seq[j] == 'G') ||
            (seq[i] == 'A' && seq[j] == 'U') || 
            (seq[i] == 'U' && seq[j] == 'A')) {
            table[i][j] = 1;
        }
    }
    
    // Fill the rest of the table
    for (int len = 2; len < n; len++) {
        for (int i = 0; i < n - len; i++) {
            int j = i + len;
            
            // Case 1: skip seq[i]
            int max_val = table[i+1][j];
            
            // Case 2: skip seq[j]
            if (table[i][j-1] > max_val) {
                max_val = table[i][j-1];
            }
            
            // Case 3: pair seq[i] and seq[j]
            int pair_score = 0;
            if ((seq[i] == 'A' && seq[j] == 'T') || 
                (seq[i] == 'T' && seq[j] == 'A') || 
                (seq[i] == 'G' && seq[j] == 'C') || 
                (seq[i] == 'C' && seq[j] == 'G') ||
                (seq[i] == 'A' && seq[j] == 'U') || 
                (seq[i] == 'U' && seq[j] == 'A')) {
                pair_score = 1;
            }
            int case3 = table[i+1][j-1] + pair_score;
            if (case3 > max_val) {
                max_val = case3;
            }
            
            // Case 4: bifurcation
            for (int k = i+1; k < j; k++) {
                int bifurcation = table[i][k] + table[k+1][j];
                if (bifurcation > max_val) {
                    max_val = bifurcation;
                }
            }
            
            table[i][j] = max_val;
        }
    }
}
