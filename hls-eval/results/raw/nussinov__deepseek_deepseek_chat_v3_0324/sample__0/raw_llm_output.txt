#include "nussinov.h"

bool is_complementary(char a, char b) {
    return (a == 'A' && (b == 'T' || b == 'U')) ||
           (a == 'T' && b == 'A') ||
           (a == 'U' && b == 'A') ||
           (a == 'G' && b == 'C') ||
           (a == 'C' && b == 'G');
}

void kernel_nussinov(char seq[60], int table[60][60]) {
    const int N = 60;

    // Initialize diagonal
    for (int i = 0; i < N; i++) {
        table[i][i] = 0;
    }

    // Fill the table diagonally
    for (int length = 1; length < N; length++) {
        for (int i = 0; i < N - length; i++) {
            int j = i + length;

            // Case 1: skip i
            int max_val = table[i+1][j];

            // Case 2: skip j
            if (table[i][j-1] > max_val) {
                max_val = table[i][j-1];
            }

            // Case 3: pair i and j
            int pair_score = is_complementary(seq[i], seq[j]) ? 1 : 0;
            int case3 = table[i+1][j-1] + pair_score;
            if (case3 > max_val) {
                max_val = case3;
            }

            // Case 4: bifurcation
            for (int k = i + 1; k < j; k++) {
                int bifurcation_score = table[i][k] + table[k+1][j];
                if (bifurcation_score > max_val) {
                    max_val = bifurcation_score;
                }
            }

            table[i][j] = max_val;
        }
    }
}
