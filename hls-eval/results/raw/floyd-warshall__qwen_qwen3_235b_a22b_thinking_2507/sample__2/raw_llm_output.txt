#include "floyd-warshall.h"

void kernel_floyd_warshall(int path[60][60]) {
    for (int k = 0; k < 60; k++) {
        for (int i = 0; i < 60; i++) {
            for (int j = 0; j < 60; j++) {
                if (path[i][k] != 999 && path[k][j] != 999) {
                    int new_dist = path[i][k] + path[k][j];
                    if (new_dist < path[i][j]) {
                        path[i][j] = new_dist;
                    }
                }
            }
        }
    }
}
