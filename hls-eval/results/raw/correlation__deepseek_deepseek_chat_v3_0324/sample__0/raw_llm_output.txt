#include "correlation.h"
#include <cmath>

void kernel_correlation(double float_n, double data[32][28], double corr[28][28], 
                     double mean[28], double stddev[28]) {
    const int M = 28;
    const int N = (int)float_n;
    
    // Calculate mean of each attribute
    for (int j = 0; j < M; j++) {
        mean[j] = 0.0;
        for (int i = 0; i < N; i++) {
            mean[j] += data[i][j];
        }
        mean[j] /= float_n;
    }
    
    // Calculate standard deviation of each attribute
    for (int j = 0; j < M; j++) {
        stddev[j] = 0.0;
        for (int i = 0; i < N; i++) {
            stddev[j] += (data[i][j] - mean[j]) * (data[i][j] - mean[j]);
        }
        stddev[j] = sqrt(stddev[j] / float_n);
        // Avoid division by zero
        if (stddev[j] <= 1e-20) {
            stddev[j] = 1.0;
        }
    }
    
    // Normalize the data
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            data[i][j] = (data[i][j] - mean[j]) / stddev[j];
        }
    }
    
    // Calculate correlation matrix
    for (int i = 0; i < M; i++) {
        for (int j = i; j < M; j++) {
            corr[i][j] = 0.0;
            for (int k = 0; k < N; k++) {
                corr[i][j] += data[k][i] * data[k][j];
            }
            corr[i][j] /= float_n;
            corr[j][i] = corr[i][j]; // Symmetric
        }
    }
    
    // Set diagonal to 1.0 (correlation of an attribute with itself)
    for (int i = 0; i < M; i++) {
        corr[i][i] = 1.0;
    }
}

1. First computes the mean of each attribute (column) in the dataset
2. Then calculates the standard deviation for each attribute
3. Normalizes the data by subtracting the mean and dividing by the standard deviation
4. Computes the correlation coefficients between all pairs of attributes using the normalized data
5. Ensures the correlation matrix is symmetric and has 1.0 on the diagonal (perfect correlation of an attribute with itself)
