        }
    }

    for (i = 0; i < 32; i++) {
        for (j = 0; j < 28; j++) {
            data[i][j] = (data[i][j] - mean[j]) / stddev[j];
        }
    }

    // Compute the correlation matrix
    for (i = 0; i < 28; i++) {
        corr[i][i] = 1.0; // Diagonal elements are 1.0 (correlation with itself)
        for (j = 0; j < i; j++) {

        }
    }
