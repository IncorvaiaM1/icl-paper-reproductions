#include "aes_aes.h"

// Galois field logarithm calculation
uint8_t gf_log(uint8_t x) {
    uint8_t y = 1, i;
    if (x == 0)
        return 0;
    for (i = 0; i < 255; i++) {
        if (y == x)
            return i;
        y = (y << 1) ^ (y >> 7 ? 0x1b : 0);
    }
    return 0;
}

// Galois field antilogarithm calculation
uint8_t gf_alog(uint8_t x) {
    uint8_t y = 1, i;
    for (i = 0; i < x; i++) {
        y = (y << 1) ^ (y >> 7 ? 0x1b : 0);
    }
    return y;
}

// Multiplicative inverse in Galois field
uint8_t gf_mulinv(uint8_t x) {
    if (x == 0)
        return 0;
    return gf_alog(255 - gf_log(x));
}

// S-box calculation
uint8_t rj_sbox(uint8_t x) {
    uint8_t y = gf_mulinv(x);
    return y ^ ((y << 1) | (y >> 7)) ^ ((y << 2) | (y >> 6)) ^ 
               ((y << 3) | (y >> 5)) ^ ((y << 4) | (y >> 4)) ^ 0x63;
}

// xtime operation
uint8_t rj_xtime(uint8_t x) {
    return (x << 1) ^ (x >> 7 ? 0x1b : 0);
}

// Substitute bytes using S-box
void aes_subBytes(uint8_t *buf) {
    for (int i = 0; i < 16; i++) {
        buf[i] = rj_sbox(buf[i]);
    }
}

// Add round key to buffer
void aes_addRoundKey(uint8_t *buf, uint8_t *key) {
    for (int i = 0; i < 16; i++) {
        buf[i] ^= key[i];
    }
}

// Shift rows according to AES specification
void aes_shiftRows(uint8_t *buf) {
    uint8_t temp;

    // Row 1: shift left by 1
    temp = buf[1];
    buf[1] = buf[5];
    buf[5] = buf[9];
    buf[9] = buf[13];
    buf[13] = temp;

    // Row 2: shift left by 2
    temp = buf[2];
    buf[2] = buf[10];
    buf[10] = temp;
    
    temp = buf[6];
    buf[6] = buf[14];
    buf[14] = temp;

    // Row 3: shift left by 3
    temp = buf[3];
    buf[3] = buf[15];
    buf[15] = buf[11];
    buf[11] = buf[7];
    buf[7] = temp;
}

// Mix columns operation
void aes_mixColumns(uint8_t *buf) {
    for (int i = 0; i < 4; i++) {
        uint8_t c0 = buf[i*4];
        uint8_t c1 = buf[i*4 + 1];
        uint8_t c2 = buf[i*4 + 2];
        uint8_t c3 = buf[i*4 + 3];
        
        uint8_t t0 = c0 ^ c1 ^ c2 ^ c3;
        uint8_t t1 = rj_xtime(c0 ^ c1);
        uint8_t t2 = rj_xtime(c1 ^ c2);
        uint8_t t3 = rj_xtime(c2 ^ c3);
        uint8_t t4 = rj_xtime(c3 ^ c0);
        
        buf[i*4]     = c0 ^ t1 ^ t0;
        buf[i*4 + 1] = c1 ^ t2 ^ t0;
        buf[i*4 + 2] = c2 ^ t3 ^ t0;
        buf[i*4 + 3] = c3 ^ t4 ^ t0;
    }
}

// Expand encryption key
void aes_expandEncKey(uint8_t *k, uint8_t *rc) {
    uint8_t t0, t1, t2, t3;
    int i;
    
    // Update first 4 bytes
    t0 = k[28];
    t1 = k[29];
    t2 = k[30];
    t3 = k[31];
    
    // Apply S-box and round constant
    k[0] ^= rj_sbox(t0) ^ *rc;
    k[1] ^= rj_sbox(t1);
    k[2] ^= rj_sbox(t2);
    k[3] ^= rj_sbox(t3);
    
    *rc = rj_xtime(*rc); // Update round constant
    
    // Update remaining bytes
    for (i = 4; i < 16; i += 4) {
        t0 = k[i+0];
        t1 = k[i+1];
        t2 = k[i+2];
        t3 = k[i+3];
        
        k[i+0] ^= k[i-4];
        k[i+1] ^= k[i-3];
        k[i+2] ^= k[i-2];
        k[i+3] ^= k[i-1];
    }
    
    // Handle last 16 bytes specially for AES-256
    for (i = 16; i < 32; i++) {
        if (i % 4 == 0) {
            k[i] ^= rj_sbox(k[i-4]);
            k[i+1] ^= rj_sbox(k[i-3]);
            k[i+2] ^= rj_sbox(k[i-2]);
            k[i+3] ^= rj_sbox(k[i-1]);
        } else {
            k[i] ^= k[i-4];
        }
    }
}

// Main AES-256 ECB encryption function
void aes256_encrypt_ecb(aes256_context *ctx, uint8_t k[32], uint8_t buf[16]) {
    uint8_t round;
    uint8_t rc = 1;
    int i;
    
    // Copy key to context
    for (i = 0; i < 32; i++) {
        ctx->key[i] = k[i];
    }
    
    // Initial round key addition
    aes_addRoundKey(buf, k);
    
    // Perform 13 rounds of encryption
    for (round = 1; round <= 13; round++) {
        // SubBytes
        aes_subBytes(buf);
        
        // ShiftRows
        aes_shiftRows(buf);
        
        // MixColumns (except last round)
        aes_mixColumns(buf);
        
        // Expand key and add round key
        if (round % 2 == 1) {
            aes_expandEncKey(ctx->key, &rc);
        }
        aes_addRoundKey(buf, &(ctx->key[((round + 1) / 2) * 16]));
    }
    
    // Final round (no MixColumns)
    aes_subBytes(buf);
    aes_shiftRows(buf);
    
    // Expand key for final round and add
    aes_expandEncKey(ctx->key, &rc);
    aes_addRoundKey(buf, &(ctx->key[224])); // 14th round key (14*16=224)
}
