#include "aes_aes.h"

// Galois field logarithm
uint8_t gf_log(uint8_t x) {
    uint8_t y = 1;
    uint8_t i;
    for (i = 0; i < 255; i++) {
        if (y == x)
            return i;
        y = (y << 1) ^ (y >> 7) * 0x1b;
    }
    return 0;
}

// Galois field anti-logarithm
uint8_t gf_alog(uint8_t x) {
    uint8_t y = 1;
    uint8_t i;
    for (i = 0; i < x; i++) {
        y = (y << 1) ^ (y >> 7) * 0x1b;
    }
    return y;
}

// Multiplicative inverse in Galois field
uint8_t gf_mulinv(uint8_t x) {
    if (x == 0)
        return 0;
    return gf_alog(255 - gf_log(x));
}

// Rijndael S-box
uint8_t rj_sbox(uint8_t x) {
    uint8_t y = gf_mulinv(x);
    y = (y << 1) | (y >> 7);
    y ^= (y << 1) | (y >> 7);
    y ^= (y << 1) | (y >> 7);
    y ^= (y << 1) | (y >> 7);
    y ^= 0x63;
    return y;
}

// Rijndael xtime
uint8_t rj_xtime(uint8_t x) {
    return (x << 1) ^ (x >> 7) * 0x1b;
}

// Substitute bytes
void aes_subBytes(uint8_t *buf) {
    uint8_t i;
    for (i = 0; i < 16; i++) {
        buf[i] = rj_sbox(buf[i]);
    }
}

// Shift rows
void aes_shiftRows(uint8_t *buf) {
    uint8_t temp;

    // Row 1: shift left by 1
    temp = buf[1];
    buf[1] = buf[5];
    buf[5] = buf[9];
    buf[9] = buf[13];
    buf[13] = temp;

    // Row 2: shift left by 2
    temp = buf[2];
    buf[2] = buf[10];
    buf[10] = temp;
    temp = buf[6];
    buf[6] = buf[14];
    buf[14] = temp;

    // Row 3: shift left by 3
    temp = buf[3];
    buf[3] = buf[15];
    buf[15] = buf[11];
    buf[11] = buf[7];
    buf[7] = temp;
}

// Mix columns
void aes_mixColumns(uint8_t *buf) {
    uint8_t i, t, u, v;
    for (i = 0; i < 16; i += 4) {
        t = buf[i] ^ buf[i+1] ^ buf[i+2] ^ buf[i+3];
        u = buf[i];
        buf[i] ^= t ^ rj_xtime(buf[i] ^ buf[i+1]);
        buf[i+1] ^= t ^ rj_xtime(buf[i+1] ^ buf[i+2]);
        buf[i+2] ^= t ^ rj_xtime(buf[i+2] ^ buf[i+3]);
        buf[i+3] ^= t ^ rj_xtime(buf[i+3] ^ u);
    }
}

// Add round key
void aes_addRoundKey(uint8_t *buf, uint8_t *key) {
    uint8_t i;
    for (i = 0; i < 16; i++) {
        buf[i] ^= key[i];
    }
}

// Expand encryption key
void aes_expandEncKey(uint8_t *k, uint8_t *rc) {
    uint8_t i;
    uint8_t temp[4];
    
    // Copy last 4 bytes to temp
    for (i = 0; i < 4; i++) {
        temp[i] = k[28+i];
    }
    
    // Apply transformation to temp
    temp[0] = rj_sbox(temp[1]) ^ (*rc);
    temp[1] = rj_sbox(temp[2]);
    temp[2] = rj_sbox(temp[3]);
    temp[3] = rj_sbox(temp[0]);
    
    // XOR with corresponding positions
    for (i = 0; i < 4; i++) {
        k[i] ^= temp[i];
    }
    
    // For each 4-byte word in the key schedule
    for (i = 4; i < 16; i++) {
        k[i] ^= k[i-4];
    }
    
    // Second half transformation
    for (i = 0; i < 4; i++) {
        temp[i] = k[12+i];
    }
    
    temp[0] = rj_sbox(temp[1]);
    temp[1] = rj_sbox(temp[2]);
    temp[2] = rj_sbox(temp[3]);
    temp[3] = rj_sbox(temp[0]);
    
    for (i = 0; i < 4; i++) {
        k[16+i] ^= temp[i];
    }
    
    for (i = 20; i < 32; i++) {
        k[i] ^= k[i-4];
    }
}

// Main AES-256 ECB encryption function
void aes256_encrypt_ecb(aes256_context *ctx, uint8_t k[32], uint8_t buf[16]) {
    uint8_t round;
    uint8_t rc[] = {0x01, 0x02, 0x04, 0x08, 0x10, 0x20, 0x40, 0x80, 0x1b, 0x36};
    
    // Initialize context key
    for (int i = 0; i < 32; i++) {
        ctx->key[i] = k[i];
    }
    
    // First add round key
    aes_addRoundKey(buf, k);
    
    // Perform 13 rounds of encryption
    for (round = 0; round < 13; round++) {
        aes_subBytes(buf);
        aes_shiftRows(buf);
        aes_mixColumns(buf);
        
        // Expand key for next round
        if (round < 7) {
            aes_expandEncKey(ctx->key, &rc[round]);
        }
        
        aes_addRoundKey(buf, &ctx->key[((round + 1) * 16) % 32]);
    }
    
    // Final round (without mixColumns)
    aes_subBytes(buf);
    aes_shiftRows(buf);
    
    // Final key expansion if needed
    if (round < 7) {
        aes_expandEncKey(ctx->key, &rc[round]);
    }
    
    aes_addRoundKey(buf, &ctx->key[16]);
}
