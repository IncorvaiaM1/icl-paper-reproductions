#include "syrk.h"

void kernel_syrk(double alpha, double beta, double C[30][30], double A[30][20]) {
    // SYRK: C = alpha * A * A^T + beta * C
    // where C is NxN symmetric matrix (lower triangular stored)
    // and A is NxM matrix
    
    // Loop indices
    int i, j, k;
    
    // Temporary matrix to store the result of A * A^T
    double temp_C[30][30];
    
    // Initialize temporary matrix
    for (i = 0; i < 30; i++) {
        for (j = 0; j < 30; j++) {
            temp_C[i][j] = 0.0;
        }
    }
    
    // Compute C = A * A^T (only lower triangular part since C is symmetric)
    for (i = 0; i < 30; i++) {
        for (j = 0; j <= i; j++) { // Only compute lower triangular part
            for (k = 0; k < 20; k++) {
                temp_C[i][j] += A[i][k] * A[j][k];
            }
        }
    }
    
    // Update C = alpha * (A * A^T) + beta * C
    for (i = 0; i < 30; i++) {
        for (j = 0; j <= i; j++) { // Only update lower triangular part
            C[i][j] = alpha * temp_C[i][j] + beta * C[i][j];
        }
    }
}
