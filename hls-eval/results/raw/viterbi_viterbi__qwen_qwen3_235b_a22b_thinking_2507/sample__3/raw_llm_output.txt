#include "viterbi_viterbi.h"

int viterbi(tok_t obs[N_OBS], prob_t init[N_STATES], prob_t transition[N_STATES * N_STATES], prob_t emission[N_STATES * N_TOKENS], state_t path[N_OBS]) {
    // Declare arrays for dynamic programming
    prob_t llike[N_OBS][N_STATES];
    state_t backptr[N_OBS][N_STATES];

    // L_init: Initialize the likelihood array for the first observation
    for (int i = 0; i < N_STATES; i++) {
        llike[0][i] = init[i] + emission[i * N_TOKENS + obs[0]];
    }

    // L_timestep: Iterate over each time step starting from the second observation
    for (int t = 1; t < N_OBS; t++) {
        // L_curr_state: Compute likelihood for each current state
        for (int j = 0; j < N_STATES; j++) {
            // L_prev_state: Find the minimum cost from previous states
            prob_t min_val = llike[t-1][0] + transition[0 * N_STATES + j];
            state_t min_state = 0;
            for (int i = 1; i < N_STATES; i++) {
                prob_t temp = llike[t-1][i] + transition[i * N_STATES + j];
                if (temp < min_val) {
                    min_val = temp;
                    min_state = i;
                }
            }
            llike[t][j] = min_val + emission[j * N_TOKENS + obs[t]];
            backptr[t][j] = min_state;
        }
    }

    // L_end: Find the best state at the final time step
    state_t best_state = 0;
    prob_t min_val = llike[N_OBS-1][0];
    for (int i = 1; i < N_STATES; i++) {
        if (llike[N_OBS-1][i] < min_val) {
            min_val = llike[N_OBS-1][i];
            best_state = i;
        }
    }
    path[N_OBS-1] = best_state;

    // L_backtrack: Recover the most likely state sequence by backtracking
    for (int t = N_OBS-2; t >= 0; t--) {
        best_state = backptr[t+1][best_state];
        path[t] = best_state;
    }

    return 0;
}
