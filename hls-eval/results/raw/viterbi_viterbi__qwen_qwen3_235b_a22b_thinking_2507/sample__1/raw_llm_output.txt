#include "viterbi_viterbi.h"

int viterbi(tok_t obs[N_OBS], prob_t init[N_STATES], prob_t transition[N_STATES * N_STATES], prob_t emission[N_STATES * N_TOKENS], state_t path[N_OBS]) {
    prob_t llike[N_OBS][N_STATES];
    state_t prev[N_OBS][N_STATES];

    for (state_t s = 0; s < N_STATES; ++s) {
        llike[0][s] = init[s] + emission[s * N_TOKENS + obs[0]];
    }

    for (step_t t = 1; t < N_OBS; ++t) {
        for (state_t s = 0; s < N_STATES; ++s) {
            prob_t candidate = llike[t-1][0] + transition[0 * N_STATES + s] + emission[s * N_TOKENS + obs[t]];
            llike[t][s] = candidate;
            prev[t][s] = 0;

            for (state_t sp = 1; sp < N_STATES; ++sp) {
                candidate = llike[t-1][sp] + transition[sp * N_STATES + s] + emission[s * N_TOKENS + obs[t]];
                if (candidate < llike[t][s]) {
                    llike[t][s] = candidate;
                    prev[t][s] = sp;
                }
            }
        }
    }

    state_t best_state = 0;
    prob_t best_prob = llike[N_OBS-1][0];
    for (state_t s = 1; s < N_STATES; ++s) {
        if (llike[N_OBS-1][s] < best_prob) {
            best_prob = llike[N_OBS-1][s];
            best_state = s;
        }
    }

    path[N_OBS-1] = best_state;
    for (step_t t = N_OBS-2; t >= 0; --t) {
        path[t] = prev[t+1][path[t+1]];
    }

    return 0;
}
