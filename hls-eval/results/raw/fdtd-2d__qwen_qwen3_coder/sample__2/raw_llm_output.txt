#include "fdtd-2d.h"

void kernel_fdtd_2d(
    double ex[20][30],
    double ey[20][30],
    double hz[20][30],
    double _fict_[20]) {

    // Coefficients as defined in PolyBench
    const double CZ = 0.3333333333333333;
    const double CEX = 0.5;
    const double CEY = 0.75;
    const double CHX = 0.6666666666666666;
    const double CHY = 1.0;

    int t, i, j;

    // Time iteration
    for (t = 0; t < 20; t++) {
        // Update Hz field
        for (i = 0; i < 20; i++) {
            for (j = 0; j < 30; j++) {
                if (j < 29 && i < 19) {
                    hz[i][j] = CZ * hz[i][j] + 
                               CEX * (ex[i][j+1] - ex[i][j] - ey[i+1][j] + ey[i][j]);
                }
            }
        }

        // Apply boundary condition for Ex
        for (i = 0; i < 20; i++) {
            ey[i][0] = _fict_[t];
        }

        // Update Ex field
        for (i = 0; i < 20; i++) {
            for (j = 1; j < 30; j++) {
                ex[i][j] = CHX * (ex[i][j] - (hz[i][j] - hz[i][j-1]));
            }
        }

        // Update Ey field
        for (i = 1; i < 20; i++) {
            for (j = 0; j < 30; j++) {
                ey[i][j] = CHY * (ey[i][j] - (hz[i][j] - hz[i-1][j]));
            }
        }
    }
}
