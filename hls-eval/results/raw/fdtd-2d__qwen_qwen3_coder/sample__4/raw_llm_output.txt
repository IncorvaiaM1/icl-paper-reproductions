#include "fdtd-2d.h"

void kernel_fdtd_2d(
    double ex[20][30],
    double ey[20][30],
    double hz[20][30],
    double _fict_[20]) {

    int t, i, j;

    // Coefficients (from original PolyBench, simplified as scalars)
    const double C_hzh = 0.999;
    const double C_hze = 0.001;
    const double C_exe = 0.999;
    const double C_exh = 0.001;
    const double C_eye = 0.999;
    const double C_eyh = 0.001;

    // Time iteration loop (for simplicity, run just one time step)
    for (t = 0; t < 20; t++) {
        // Update Hz
        for (i = 0; i < 20 - 1; i++)
            for (j = 0; j < 30 - 1; j++) {
                hz[i][j] = C_hzh * hz[i][j] +
                           C_hze * (ex[i][j+1] - ex[i][j] - ey[i+1][j] + ey[i][j]);
            }

        // Apply source to Ez (Ex and Ey fields get updated accordingly)
        for (i = 0; i < 20; i++) {
            ey[i][0] = _fict_[t];
        }

        // Update Ex
        for (i = 0; i < 20; i++)
            for (j = 1; j < 30; j++) {
                ex[i][j] = C_exe * ex[i][j] +
                           C_exh * (hz[i][j] - hz[i][j-1]);
            }

        // Update Ey
        for (i = 1; i < 20; i++)
            for (j = 0; j < 30; j++) {
                ey[i][j] = C_eye * ey[i][j] +
                           C_eyh * (hz[i][j] - hz[i-1][j]);
            }
    }
}
