#include "fdtd-2d.h"

void kernel_fdtd_2d(
    double ex[20][30],
    double ey[20][30],
    double hz[20][30],
    double _fict_[20]) {

    // Coefficient definitions
    const double C_hzh = 0.999;
    const double C_hze = 0.001;
    const double C_exe = 0.999;
    const double C_exh = 0.001;
    const double C_eye = 0.999;
    const double C_eyh = 0.001;

    // Main computation loop
    for (int t = 0; t < 20; t++) {
        // Update Hz field
        for (int i = 0; i < 20; i++) {
            for (int j = 0; j < 30; j++) {
                double dEx_dy = (j < 29) ? (ex[i][j+1] - ex[i][j]) : 0.0;
                double dEy_dx = (i < 19) ? (ey[i+1][j] - ey[i][j]) : 0.0;
                hz[i][j] = C_hzh * hz[i][j] + C_hze * (dEx_dy - dEy_dx);
            }
        }

        // Update Ex field
        for (int i = 0; i < 20; i++) {
            for (int j = 0; j < 30; j++) {
                double dHz_dy = (j > 0) ? (hz[i][j] - hz[i][j-1]) : 0.0;
                ex[i][j] = C_exe * ex[i][j] + C_exh * dHz_dy;
            }
        }

        // Update Ey field
        for (int i = 0; i < 20; i++) {
            for (int j = 0; j < 30; j++) {
                double dHz_dx = (i > 0) ? (hz[i][j] - hz[i-1][j]) : 0.0;
                ey[i][j] = C_eye * ey[i][j] + C_eyh * dHz_dx;
            }
        }

        // Apply fictitious source to Ey's first column (j=0)
        for (int i = 0; i < 20; i++) {
            ey[i][0] = _fict_[t];
        }
    }
}
