#include "fdtd-2d.h"

void kernel_fdtd_2d(
    double ex[20][30],
    double ey[20][30],
    double hz[20][30],
    double _fict_[20]) {

    // Coefficients as per PolyBench simplification
    const double C_hzh = 0.999;
    const double C_hze = 0.001;
    const double C_exe = 0.999;
    const double C_exh = 0.001;
    const double C_eye = 0.999;
    const double C_eyh = 0.001;

    // Time iteration loop (simplified to one step as per typical PolyBench structure)
    int t = 0;

    // Update Hz field
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            double tmp_ex = 0.0;
            double tmp_ey = 0.0;

            if (j + 1 < 30) {
                tmp_ex = ex[i][j + 1];
            }

            if (i + 1 < 20) {
                tmp_ey = ey[i + 1][j];
            }

            hz[i][j] = C_hzh * hz[i][j] +
                       C_hze * (tmp_ex - ex[i][j] - tmp_ey + ey[i][j]);
        }
    }

    // Update Ex field
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            double tmp_hz_prev = 0.0;
            if (j - 1 >= 0) {
                tmp_hz_prev = hz[i][j - 1];
            } else if (i == 0) {
                tmp_hz_prev = _fict_[t];
            }

            ex[i][j] = C_exe * ex[i][j] +
                       C_exh * (hz[i][j] - tmp_hz_prev);
        }
    }

    // Update Ey field
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            double tmp_hz_prev = 0.0;
            if (i - 1 >= 0) {
                tmp_hz_prev = hz[i - 1][j];
            }

            ey[i][j] = C_eye * ey[i][j] +
                       C_eyh * (hz[i][j] - tmp_hz_prev);
        }
    }
}
