// Helper function to rotate a 80-bit key left by 61 bits
static void rotateLeft61(present_key_t key) {
    unsigned char temp[10];
    // 61 bits rotation is equivalent to 61 % 80 = 61, which is 7 full bytes and 5 bits
    // Save the upper 5 bits of the first byte
    unsigned char carry = (key[0] >> 3) & 0x1F;
    
    // Rotate each byte
    for (int i = 0; i < 9; i++) {
        temp[i] = (key[i] << 5) | (key[i+1] >> 3);
    }
    // Handle the last byte
    temp[9] = (key[9] << 5) | carry;
    
    // Copy back
    for (int i = 0; i < 10; i++) {
        key[i] = temp[i];
    }
}

// Key generation function for 80-bit key
void generateRoundKeys80(present_key_t *suppliedKey, keys_t *keys) {
    present_key_t key;
    copyKey(suppliedKey, &key);
    
    for (int i = 0; i < ROUNDS; i++) {
        // The first 64 bits of the current key are the round key
        for (int j = 0; j < ROUND_KEY_SIZE_BYTES; j++) {
            (*keys)[i][j] = key[j];
        }
        
        // Key update
        // 1. Rotate left by 61 bits
        rotateLeft61(key);
        
        // 2. S-box application on the leftmost 4 bits (nibble)
        unsigned char nibble = (key[0] >> 4) & 0x0F;
        nibble = sBox[nibble];
        key[0] = (key[0] & 0x0F) | (nibble << 4);
        
        // 3. XOR with round counter (i+1) on bits 15-19 (bits 15-19 are in bytes 1 and 2)
        unsigned char round = i + 1;
        key[1] ^= (round >> 3) & 0xFF;
        key[2] ^= (round << 5) & 0xE0;
    }
}

// Add round key to the block
void addRoundKey(block_t *block, round_key_t *roundKey) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        (*block)[i] ^= (*roundKey)[i];
    }
}

// Substitution layer (S-box applied to each nibble)
void sBoxLayer(block_t *block) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        // Upper nibble
        unsigned char nibble = ((*block)[i] >> 4) & 0x0F;
        nibble = sBox[nibble];
        // Lower nibble
        unsigned char lowerNibble = (*block)[i] & 0x0F;
        lowerNibble = sBox[lowerNibble];
        // Combine
        (*block)[i] = (nibble << 4) | lowerNibble;
    }
}

// Permutation layer
void pLayer(block_t *block) {
    block_t temp;
    for (int i = 0; i < 64; i++) {
        unsigned char bit = ((*block)[i / 8] >> (7 - (i % 8))) & 0x01;
        if (bit) {
            int newPos = (i * 16) % 63;
            if (i == 63) newPos = 63;
            temp[newPos / 8] |= (1 << (7 - (newPos % 8)));
        }
    }
    copyBlock(&temp, block);
}

// Copy key
void copyKey(present_key_t *from, present_key_t *to) {
    for (int i = 0; i < PRESENT_80_KEY_SIZE_BYTES; i++) {
        (*to)[i] = (*from)[i];
    }
}

// Copy block
void copyBlock(block_t *from, block_t *to) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        (*to)[i] = (*from)[i];
    }
}

// Main encryption function
void present80_encryptBlock(block_t *block, present_key_t *key) {
    keys_t roundKeys;
    generateRoundKeys80(key, &roundKeys);
    
    for (int i = 0; i < ROUNDS - 1; i++) {
        addRoundKey(block, &roundKeys[i]);
        sBoxLayer(block);
        pLayer(block);
    }
    // Final round
    addRoundKey(block, &roundKeys[ROUNDS - 1]);
}
