#include "gramschmidt.h"

void kernel_gramschmidt(double A[20][30], double R[30][30], double Q[20][30]) {
    int i, j, k;
    double norm;

    // Modified Gram-Schmidt algorithm
    for (k = 0; k < 30; k++) {
        // Compute the norm of the k-th column of A
        norm = 0.0;
        for (i = 0; i < 20; i++) {
            norm += A[i][k] * A[i][k];
        }
        norm = sqrt(norm);

        // Store the norm in the diagonal element of R
        R[k][k] = norm;

        // Normalize the k-th column of A to get the k-th column of Q
        for (i = 0; i < 20; i++) {
            Q[i][k] = A[i][k] / norm;
        }

        // Update the remaining columns of A
        for (j = k + 1; j < 30; j++) {
            // Compute the dot product of Q[:,k] and A[:,j]
            double dot_product = 0.0;
            for (i = 0; i < 20; i++) {
                dot_product += Q[i][k] * A[i][j];
            }

            // Store the dot product in R[k][j]
            R[k][j] = dot_product;

            // Subtract the projection from A[:,j]
            for (i = 0; i < 20; i++) {
                A[i][j] -= Q[i][k] * dot_product;
            }
        }
    }
}
