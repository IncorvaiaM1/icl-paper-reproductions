void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {
    // Iterate over each row of the sparse matrix
ellpack_1:
    for (int i = 0; i < N; i++) {
        TYPE sum = 0.0;
        // Iterate over each non-zero element in the current row
ellpack_2:
        for (int j = 0; j < L; j++) {
            // Compute the product of the non-zero value with the corresponding vector element
            sum += nzval[i * L + j] * vec[cols[i * L + j]];
        }
        // Store the result for the current row
        out[i] = sum;
    }
