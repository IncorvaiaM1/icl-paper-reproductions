#include "spmv_ellpack.h"

void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {
    int i, j;
    TYPE sum;

    ellpack_1:
    for (i = 0; i < N; i++) {
        sum = 0.0;

        ellpack_2:
        for (j = 0; j < L; j++) {
            sum += nzval[i * L + j] * vec[cols[i * L + j]];
        }

        out[i] = sum;
    }
}
