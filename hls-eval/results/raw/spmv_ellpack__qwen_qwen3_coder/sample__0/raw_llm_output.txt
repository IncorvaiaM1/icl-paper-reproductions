#include "spmv_ellpack.h"

void ellpack(TYPE nzval[N * L], int32_t cols[N * L], TYPE vec[N], TYPE out[N]) {
    // Iterate over each row of the sparse matrix
ellpack_1:
    for (int i = 0; i < N; i++) {
        TYPE sum = 0.0;

        // Iterate over non-zero elements in the current row
ellpack_2:
        for (int j = 0; j < L; j++) {
            int index = i * L + j;
            int col_index = cols[index];
            sum += nzval[index] * vec[col_index];
        }

        // Store the computed sum in the output vector
        out[i] = sum;
    }
}
