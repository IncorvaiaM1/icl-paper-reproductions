#include "syr2k.h"

void kernel_syr2k(
    double alpha,
    double beta,
    double C[30][30],
    double A[30][20],
    double B[30][20])
{
    // Temporary variable for storing intermediate results
    double temp;

    // Perform symmetric rank 2k update (C = alpha*AB^T + alpha*BA^T + beta*C)
    // Only updating the lower triangular part (UPLO='L')
    for (int i = 0; i < 30; i++) {
        for (int j = 0; j <= i; j++) {  // Lower triangular loop
            temp = 0.0;
            // Compute alpha*(A*B^T + B*A^T) for this element
            for (int k = 0; k < 20; k++) {  // M dimension
                temp += alpha * (A[i][k] * B[j][k] + B[i][k] * A[j][k]);
            }
            // Add beta*C and store back in C
            C[i][j] = temp + beta * C[i][j];
        }
    }
}
