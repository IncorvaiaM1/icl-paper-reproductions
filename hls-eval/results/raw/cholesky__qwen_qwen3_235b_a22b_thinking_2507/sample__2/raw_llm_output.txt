#include "cholesky.h"

void kernel_cholesky(double A[40][40]) {
    for (int i = 0; i < 40; i++) {
        for (int j = 0; j <= i; j++) {
            double sum = 0.0;
            for (int k = 0; k < j; k++) {
                sum += A[i][k] * A[j][k];
            }
            if (i == j) {
                A[i][i] = sqrt(A[i][i] - sum);
            } else {
                A[i][j] = (A[i][j] - sum) / A[j][j];
            }
        }
    }
}
