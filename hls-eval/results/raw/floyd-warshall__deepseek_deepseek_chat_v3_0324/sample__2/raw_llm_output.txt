#include "floyd-warshall.h"
#include <algorithm>

void kernel_floyd_warshall(int path[60][60]) {
    #pragma HLS INTERFACE m_axi port=path offset=slave bundle=gmem
    #pragma HLS INTERFACE s_axilite port=path bundle=control
    #pragma HLS INTERFACE s_axilite port=return bundle=control

    int temp[60][60];

    // Initialize temp matrix with path values
    for (int i = 0; i < 60; i++) {
        #pragma HLS PIPELINE II=1
        for (int j = 0; j < 60; j++) {
            #pragma HLS UNROLL factor=4
            temp[i][j] = path[i][j];
        }
    }

    // Floyd-Warshall algorithm
    for (int k = 0; k < 60; k++) {
        #pragma HLS LOOP_FLATTEN off
        for (int i = 0; i < 60; i++) {
            #pragma HLS PIPELINE II=1
            for (int j = 0; j < 60; j++) {
                #pragma HLS UNROLL factor=4
                int new_path = temp[i][k] + temp[k][j];
                if (temp[i][j] > new_path) {
                    temp[i][j] = new_path;
                }
            }
        }
    }

    // Copy results back to output
    for (int i = 0; i < 60; i++) {
        #pragma HLS PIPELINE II=1
        for (int j = 0; j < 60; j++) {
            #pragma HLS UNROLL factor=4
            path[i][j] = temp[i][j];
        }
    }
}
