#include "bfs_queue.h"

void bfs(
    node_t nodes[N_NODES],
    edge_t edges[N_EDGES],
    node_index_t starting_node,
    level_t level[N_NODES],
    edge_index_t level_counts[N_LEVELS]) {
    
    // Queue implementation using circular buffer
    node_index_t queue[N_NODES];
    int queue_head = 0;
    int queue_tail = 0;
    int queue_size = 0;
    
    // Initialize levels to MAX_LEVEL
    INIT_LEVELS: for (int i = 0; i < N_NODES; i++) {
        level[i] = MAX_LEVEL;
    }
    
    // Initialize level counts to 0
    INIT_HORIZONS: for (int i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }
    
    // Set starting node level to 0 and add to queue
    level[starting_node] = 0;
    level_counts[0] = 1;
    queue[queue_tail] = starting_node;
    queue_tail = (queue_tail + 1) % N_NODES;
    queue_size++;
    
    int current_level = 0;
    
    // Process nodes in queue level by level
    LOOP_QUEUE: while (queue_size > 0 && current_level < N_LEVELS - 1) {
        int nodes_at_current_level = queue_size;
        
        // Process all nodes at the current level
        for (int i = 0; i < nodes_at_current_level; i++) {
            // Dequeue node
            node_index_t current_node = queue[queue_head];
            queue_head = (queue_head + 1) % N_NODES;
            queue_size--;
            
            // Process neighbors of current node
            edge_index_t edge_start = nodes[current_node].edge_begin;
            edge_index_t edge_end = nodes[current_node].edge_end;
            
            LOOP_NEIGHBORS: for (edge_index_t e = edge_start; e < edge_end; e++) {
                node_index_t neighbor = edges[e].dst;
                
                // If neighbor hasn't been visited yet
                if (level[neighbor] == MAX_LEVEL) {
                    level[neighbor] = current_level + 1;
                    level_counts[current_level + 1]++;
                    // Enqueue neighbor
                    queue[queue_tail] = neighbor;
                    queue_tail = (queue_tail + 1) % N_NODES;
                    queue_size++;
                }
            }
        }
        current_level++;
    }
}
