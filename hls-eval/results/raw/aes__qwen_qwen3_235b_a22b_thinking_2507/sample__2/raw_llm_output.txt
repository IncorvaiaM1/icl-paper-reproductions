#include "aes.h"

static void AddRoundKey(uint8_t round, state_t *state, const round_t *RoundKey) {
    uint8_t *rkey = (uint8_t *)RoundKey;
    uint8_t offset = round * 16;
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            (*state)[i][j] ^= rkey[offset + i*4 + j];
        }
    }
}

static void SubBytes(state_t *state) {
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            (*state)[i][j] = sbox[(*state)[i][j]];
        }
    }
}

static void ShiftRows(state_t *state) {
    // Row 1: shift left by 1
    uint8_t temp = (*state)[1][0];
    (*state)[1][0] = (*state)[1][1];
    (*state)[1][1] = (*state)[1][2];
    (*state)[1][2] = (*state)[1][3];
    (*state)[1][3] = temp;
    
    // Row 2: shift left by 2
    temp = (*state)[2][0];
    (*state)[2][0] = (*state)[2][2];
    (*state)[2][2] = temp;
    temp = (*state)[2][1];
    (*state)[2][1] = (*state)[2][3];
    (*state)[2][3] = temp;
    
    // Row 3: shift left by 3 (equivalent to shift right by 1)
    temp = (*state)[3][3];
    (*state)[3][3] = (*state)[3][2];
    (*state)[3][2] = (*state)[3][1];
    (*state)[3][1] = (*state)[3][0];
    (*state)[3][0] = temp;
}

static void MixColumns(state_t *state) {
    for (int j = 0; j < 4; j++) {
        uint8_t a0 = (*state)[0][j];
        uint8_t a1 = (*state)[1][j];
        uint8_t a2 = (*state)[2][j];
        uint8_t a3 = (*state)[3][j];
        
        uint8_t t = a0 ^ a1 ^ a2 ^ a3;
        uint8_t x = a0 ^ a1;
        x = (x & 0x80) ? (x << 1) ^ 0x1b : x << 1;
        uint8_t y = a2 ^ a3;
        y = (y & 0x80) ? (y << 1) ^ 0x1b : y << 1;
        
        (*state)[0][j] = a0 ^ x ^ t;
        (*state)[1][j] = a1 ^ x ^ t;
        (*state)[2][j] = a2 ^ y ^ t;
        (*state)[3][j] = a3 ^ y ^ t;
    }
}

void Cipher(state_t *state, const round_t *RoundKey) {
    AddRoundKey(0, state, RoundKey);
    
    for (uint8_t round = 1; round < Nr; round++) {
        SubBytes(state);
        ShiftRows(state);
        MixColumns(state);
        AddRoundKey(round, state, RoundKey);
    }
    
    SubBytes(state);
    ShiftRows(state);
    AddRoundKey(Nr, state, RoundKey);
}
