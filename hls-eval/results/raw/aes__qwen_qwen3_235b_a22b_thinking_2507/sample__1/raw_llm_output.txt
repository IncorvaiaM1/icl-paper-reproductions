#include "aes.h"

void AddRoundKey(uint8_t round, state_t *state, const round_t *RoundKey) {
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            (*state)[i][j] ^= RoundKey[round * 16 + i * 4 + j];
        }
    }
}

void SubBytes(state_t *state) {
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            (*state)[i][j] = sbox[(*state)[i][j]];
        }
    }
}
