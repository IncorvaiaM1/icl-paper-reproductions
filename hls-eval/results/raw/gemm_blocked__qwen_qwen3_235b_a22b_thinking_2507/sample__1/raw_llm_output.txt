#include "gemm_blocked.h"

void bbgemm(TYPE m1[N], TYPE m2[N], TYPE prod[N]) {
    // Initialize the output matrix to zero
    for (int i = 0; i < N; i++) {
        prod[i] = 0;
    }

    // Blocked GEMM implementation with 5 nested loops
    // Outer loops iterate over blocks (jj and kk)
    // Middle loops iterate over rows (i) and inner product dimension (k)
    // Innermost loop accumulates partial products over columns (j)
    for (int loopjj = 0; loopjj < col_size; loopjj += block_size) {
        for (int loopkk = 0; loopkk < row_size; loopkk += block_size) {
            for (int loopi = 0; loopi < row_size; loopi++) {
                for (int loopk = loopkk; loopk < loopkk + block_size && loopk < row_size; loopk++) {
                    for (int loopj = loopjj; loopj < loopjj + block_size && loopj < col_size; loopj++) {
                        // Compute matrix multiplication for current block
                        // Access matrices in row-major order
                        prod[loopi * col_size + loopj] += 
                            m1[loopi * col_size + loopk] * 
                            m2[loopk * col_size + loopj];
                    }
                }
            }
        }
    }
}
