#include "spmv_crs.h"

void spmv(
    TYPE val[NNZ],
    int32_t cols[NNZ],
    int32_t rowDelimiters[N + 1],
    TYPE vec[N],
    TYPE out[N])
{
    // Initialize output vector to zero
    for (int i = 0; i < N; i++) {
        out[i] = 0;
    }

    // Main computation loop
    ROW_LOOP:
    for (int i = 0; i < N; i++) {
        TYPE sum = 0;
        int row_start = rowDelimiters[i];
        int row_end = rowDelimiters[i + 1];

        // Compute dot product for current row
        NNZ_LOOP:
        for (int j = row_start; j < row_end; j++) {
            sum += val[j] * vec[cols[j]];
        }

        out[i] = sum;
    }
}
