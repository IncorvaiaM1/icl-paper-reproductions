#include "present.h"

static void addRoundKey(block_t *block, round_key_t *roundKey) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        (*block)[i] ^= (*roundKey)[i];
    }
}

static void substituteLayer(block_t *block) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        unsigned char high = (*block)[i] >> 4;
        unsigned char low = (*block)[i] & 0x0F;
        (*block)[i] = (sBox[high] << 4) | sBox[low];
    }
}

static void pLayer(block_t *block) {
    block_t temp = {0};
    for (int j = 0; j < 64; j++) {
        int i = 4 * (j % 16) + (j / 16);
        int byte_in = i / 8;
        int bit_in = 7 - (i % 8);
        int bit_val = ((*block)[byte_in] >> bit_in) & 1;
        int byte_out = j / 8;
        int bit_out = 7 - (j % 8);
        if (bit_val) {
            temp[byte_out] |= (1 << bit_out);
        }
    }
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        (*block)[i] = temp[i];
    }
}

static void generateRoundKeys80(present_key_t *suppliedKey, keys_t *keys) {
    present_key_t K;
    for (int i = 0; i < PRESENT_80_KEY_SIZE_BYTES; i++) {
        K[i] = (*suppliedKey)[i];
    }

    for (int i = 0; i < ROUNDS; i++) {
        for (int j = 0; j < ROUND_KEY_SIZE_BYTES; j++) {
            (*keys)[i][j] = K[j];
        }

        if (i < ROUNDS - 1) {
            unsigned char left4 = K[0] >> 4;
            unsigned char s = sBox[left4];
            K[0] = (s << 4) | (K[0] & 0x0F);

            unsigned char t = K[0];
            for (int j = 0; j < 9; j++) {
                K[j] = (K[j] << 5) | (K[j+1] >> 3);
            }
            K[9] = (K[9] << 5) | (t >> 3);

            unsigned char c = i + 1;
            unsigned char mask1 = c & 0x01;
            unsigned char mask2 = 0;
            if (c & 0x02) mask2 |= 0x80;
            if (c & 0x04) mask2 |= 0x40;
            if (c & 0x08) mask2 |= 0x20;
            if (c & 0x10) mask2 |= 0x10;
            
            K[1] ^= mask1;
            K[2] ^= mask2;
        }
    }
}

void present80_encryptBlock(block_t *block, present_key_t *key) {
    keys_t keys;
    generateRoundKeys80(key, &keys);

    for (int i = 0; i < 31; i++) {
        addRoundKey(block, &keys[i]);
        substituteLayer(block);
        pLayer(block);
    }
    
    addRoundKey(block, &keys[31]);
    substituteLayer(block);
}
