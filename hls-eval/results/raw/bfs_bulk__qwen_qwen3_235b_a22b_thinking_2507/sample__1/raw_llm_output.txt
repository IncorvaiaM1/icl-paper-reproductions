#include "bfs_bulk.h"

void bfs(node_t nodes[N_NODES], edge_t edges[N_EDGES], node_index_t starting_node, level_t level[N_NODES], edge_index_t level_counts[N_LEVELS]) {
    // Initialize level counts to zero
    for (int i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }
    level_counts[0] = 1;
    level[starting_node] = 0;

    // Iterate through each level (horizon)
    for (level_t horizon = 0; horizon < N_LEVELS; horizon++) {
        // Iterate through all nodes
        for (node_index_t n = 0; n < N_NODES; n++) {
            // Check if current node is at current horizon
            if (level[n] == horizon) {
                // Get edge range for this node
                edge_index_t tmp_begin = nodes[n].edge_begin;
                edge_index_t tmp_end = nodes[n].edge_end;

                // Process each edge
                for (edge_index_t e = tmp_begin; e < tmp_end; e++) {
                    node_index_t dst = edges[e].dst;
                    // If destination node is unvisited
                    if (level[dst] == MAX_LEVEL) {
                        // Mark it as visited at the next level
                        level[dst] = horizon + 1;
                        // Only increment if next level is within bounds
                        if ((horizon + 1) < N_LEVELS) {
                            level_counts[horizon + 1]++;
                        }
                    }
                }
            }
        }
    }
}
