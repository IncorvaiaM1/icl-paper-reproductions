#include "gesummv.h"

void kernel_gesummv(double alpha, double beta, double A[30][30], double B[30][30], double tmp[30], double x[30], double y[30]) {
    // Compute A * x and store in tmp
    for (int i = 0; i < 30; i++) {
        double sum_a = 0.0;
        for (int j = 0; j < 30; j++) {
            sum_a += A[i][j] * x[j];
        }
        tmp[i] = sum_a;
    }

    // Compute B * x and store in y
    for (int i = 0; i < 30; i++) {
        double sum_b = 0.0;
        for (int j = 0; j < 30; j++) {
            sum_b += B[i][j] * x[j];
        }
        y[i] = sum_b;
    }

    // Compute final result: y = alpha * tmp + beta * y
    for (int i = 0; i < 30; i++) {
        y[i] = alpha * tmp[i] + beta * y[i];
    }
}
