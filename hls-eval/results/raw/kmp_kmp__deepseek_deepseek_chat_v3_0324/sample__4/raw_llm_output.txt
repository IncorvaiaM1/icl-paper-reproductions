#include "kmp_kmp.h"

void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    int k, q;
    kmpNext[0] = -1;
    k = -1;
    for (q = 1; q < PATTERN_SIZE; q++) {
        while (k >= 0 && pattern[k + 1] != pattern[q]) {
            k = kmpNext[k];
        }
        if (pattern[k + 1] == pattern[q]) {
            k++;
        }
        kmpNext[q] = k;
    }
}

int kmp(
    char pattern[PATTERN_SIZE],
    char input[STRING_SIZE],
    int32_t kmpNext[PATTERN_SIZE],
    int32_t n_matches[1]) {

    CPF(pattern, kmpNext);

    int matches = 0;
    int i = 0;
    int j = -1;
    int m = PATTERN_SIZE;
    int n = STRING_SIZE;

    for (i = 0; i < n; i++) {
        while (j >= 0 && pattern[j + 1] != input[i]) {
            j = kmpNext[j];
        }
        if (pattern[j + 1] == input[i]) {
            j++;
        }
        if (j == m - 1) {
            matches++;
            j = kmpNext[j];
        }
    }

    n_matches[0] = matches;
    return matches;
}
