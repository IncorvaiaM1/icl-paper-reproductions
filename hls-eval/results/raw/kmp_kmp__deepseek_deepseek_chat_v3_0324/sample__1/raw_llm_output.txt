#include "kmp_kmp.h"

void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    int k = -1;
    kmpNext[0] = -1;
    
    for (int q = 1; q < PATTERN_SIZE; q++) {
        while (k >= 0 && pattern[k + 1] != pattern[q]) {
            k = kmpNext[k];
        }
        if (pattern[k + 1] == pattern[q]) {
            k++;
        }
        kmpNext[q] = k;
    }
}

int kmp(char pattern[PATTERN_SIZE], char input[STRING_SIZE], int32_t kmpNext[PATTERN_SIZE], int32_t n_matches[1]) {
    #pragma HLS INTERFACE m_axi port=pattern offset=slave bundle=gmem
    #pragma HLS INTERFACE m_axi port=input offset=slave bundle=gmem
    #pragma HLS INTERFACE m_axi port=kmpNext offset=slave bundle=gmem
    #pragma HLS INTERFACE m_axi port=n_matches offset=slave bundle=gmem
    #pragma HLS INTERFACE s_axilite port=pattern bundle=control
    #pragma HLS INTERFACE s_axilite port=input bundle=control
    #pragma HLS INTERFACE s_axilite port=kmpNext bundle=control
    #pragma HLS INTERFACE s_axilite port=n_matches bundle=control
    #pragma HLS INTERFACE s_axilite port=return bundle=control

    int matches = 0;
    int q = -1;
    
    // Preprocess the pattern to build kmpNext table
    CPF(pattern, kmpNext);
    
    // Search for the pattern in input string
    for (int i = 0; i < STRING_SIZE && input[i] != '\0'; i++) {
        while (q >= 0 && pattern[q + 1] != input[i]) {
            q = kmpNext[q];
        }
        if (pattern[q + 1] == input[i]) {
            q++;
        }
        if (q == PATTERN_SIZE - 1) {
            matches++;
            q = kmpNext[q];
        }
    }
    
    n_matches[0] = matches;
    return matches;
}
