#include "kmp_kmp.h"

void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    int k, q;
    
    kmpNext[0] = -1;
    for (q = 1; q < PATTERN_SIZE; q++) {
        k = kmpNext[q - 1];
        while (k >= 0 && pattern[k] != pattern[q - 1]) {
            k = kmpNext[k];
        }
        kmpNext[q] = k + 1;
    }
}

int kmp(
    char pattern[PATTERN_SIZE],
    char input[STRING_SIZE],
    int32_t kmpNext[PATTERN_SIZE],
    int32_t n_matches[1]) {
    
    // Preprocess the pattern to build the KMP table
    CPF(pattern, kmpNext);
    
    int i = 0; // index for input[]
    int q = 0; // index for pattern[]
    n_matches[0] = 0;
    
    while (i < STRING_SIZE && input[i] != 0) {
        while (q >= 0 && pattern[q] != input[i]) {
            q = kmpNext[q];
        }
        q++;
        i++;
        if (q == PATTERN_SIZE) {
            n_matches[0]++;
            q = kmpNext[q - 1] + 1;
        }
    }
    
    return n_matches[0];
}
