#include "bfs_bulk.h"

void bfs(
    node_t nodes[N_NODES],
    edge_t edges[N_EDGES],
    node_index_t starting_node,
    level_t level[N_NODES],
    edge_index_t level_counts[N_LEVELS]) {

    // Initialize level array: set all levels to MAX_LEVEL except starting node
    level_t curr_level;
    node_index_t n;
    edge_index_t e;
    edge_index_t tmp_begin, tmp_end;

    for (n = 0; n < N_NODES; n++) {
        level[n] = MAX_LEVEL;
    }
    level[starting_node] = 0;

    // Initialize level_counts array
    for (int i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }
    level_counts[0] = 1; // Starting node is at level 0

    // Loop over levels (horizons)
    for (int horizon = 0; horizon < N_LEVELS - 1; horizon++) {
        // Exit early if no nodes at this level
        if (level_counts[horizon] == 0) {
            break;
        }

        // Process all nodes at current level
        for (n = 0; n < N_NODES; n++) {
            if (level[n] == horizon) {
                // Get the edge range for this node
                tmp_begin = nodes[n].edge_begin;
                tmp_end = nodes[n].edge_end;

                // Process all neighbors of this node
                for (e = tmp_begin; e < tmp_end; e++) {
                    node_index_t neighbor = edges[e].dst;
                    // If neighbor hasn't been visited yet
                    if (level[neighbor] == MAX_LEVEL) {
                        level[neighbor] = horizon + 1;
                        level_counts[horizon + 1]++;
                    }
                }
            }
        }
    }
}
