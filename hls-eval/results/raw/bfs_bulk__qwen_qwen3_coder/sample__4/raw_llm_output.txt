#include "bfs_bulk.h"

void bfs(
    node_t nodes[N_NODES],
    edge_t edges[N_EDGES],
    node_index_t starting_node,
    level_t level[N_NODES],
    edge_index_t level_counts[N_LEVELS]) {

    // Initialize level array with MAX_LEVEL
    INIT_LEVELS:
    for (node_index_t n = 0; n < N_NODES; n++) {
        level[n] = MAX_LEVEL;
    }

    // Set the starting node's level to 0
    level[starting_node] = 0;

    // Initialize level_counts array to 0
    INIT_COUNTS:
    for (int h = 0; h < N_LEVELS; h++) {
        level_counts[h] = 0;
    }

    // Set count for level 0 to 1 (the starting node)
    level_counts[0] = 1;

    // Loop over horizons/levels
    LOOP_HORIZONS:
    for (int horizon = 0; horizon < N_LEVELS - 1; horizon++) {
        // If no nodes in this level, stop processing
        if (level_counts[horizon] == 0)
            break;

        // Iterate through all nodes to find those at the current horizon
        LOOP_NODES:
        for (node_index_t n = 0; n < N_NODES; n++) {
            // Check if the node is at the current horizon level
            if (level[n] == horizon) {
                // Get the range of edges for this node
                edge_index_t tmp_begin = nodes[n].edge_begin;
                edge_index_t tmp_end = nodes[n].edge_end;

                // Iterate through all neighbors of this node
                LOOP_NEIGHBORS:
                for (edge_index_t e = tmp_begin; e < tmp_end; e++) {
                    node_index_t neighbor = edges[e].dst;

                    // If neighbor has not been visited yet
                    if (level[neighbor] == MAX_LEVEL) {
                        // Assign it to the next horizon level
                        level[neighbor] = horizon + 1;
                        // Increment the count for the next horizon
                        level_counts[horizon + 1]++;
                    }
                }
            }
        }
    }
}
