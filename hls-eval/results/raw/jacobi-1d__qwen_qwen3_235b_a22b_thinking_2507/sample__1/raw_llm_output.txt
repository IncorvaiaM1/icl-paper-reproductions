#include "jacobi-1d.h"

void kernel_jacobi_1d(double A[30], double B[30]) {
    const int n = 30;
    const int tsteps = 20;

    for (int t = 0; t < tsteps; t++) {
        if (t % 2 == 0) {
            // Update from A (current state) to B (next state)
            B[0] = (2.0 * A[0] + A[1]) / 3.0;
            #pragma HLS PIPELINE
            for (int i = 1; i < n-1; i++) {
                B[i] = (A[i-1] + A[i] + A[i+1]) / 3.0;
            }
            B[n-1] = (A[n-2] + 2.0 * A[n-1]) / 3.0;
        } else {
            // Update from B (current state) to A (next state)
            A[0] = (2.0 * B[0] + B[1]) / 3.0;
            #pragma HLS PIPELINE
            for (int i = 1; i < n-1; i++) {
                A[i] = (B[i-1] + B[i] + B[i+1]) / 3.0;
            }
            A[n-1] = (B[n-2] + 2.0 * B[n-1]) / 3.0;
        }
    }
}
