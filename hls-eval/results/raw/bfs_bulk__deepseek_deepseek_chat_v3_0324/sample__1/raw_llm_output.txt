#include "bfs_bulk.h"

void bfs(
    node_t nodes[N_NODES],
    edge_t edges[N_EDGES],
    node_index_t starting_node,
    level_t level[N_NODES],
    edge_index_t level_counts[N_LEVELS])
{
    // Initialize level array with MAX_LEVEL (unvisited)
    for (node_index_t i = 0; i < N_NODES; i++) {
        level[i] = MAX_LEVEL;
    }

    // Initialize level counts to 0
    for (edge_index_t i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }

    // Mark starting node as level 0
    level[starting_node] = 0;
    level_counts[0] = 1;

    // Main BFS loop
    for (edge_index_t horizon = 0; horizon < N_LEVELS; horizon++) {
        // Early exit if no nodes in current level
        if (level_counts[horizon] == 0) {
            break;
        }

        // Process all nodes in current level
        for (node_index_t n = 0; n < N_NODES; n++) {
            if (level[n] == horizon) {
                // Get edge range for current node
                edge_index_t tmp_begin = nodes[n].edge_begin;
                edge_index_t tmp_end = nodes[n].edge_end;

                // Process all neighbors
                for (edge_index_t e = tmp_begin; e < tmp_end; e++) {
                    node_index_t dst = edges[e].dst;

                    // If neighbor hasn't been visited, mark it for next level
                    if (level[dst] == MAX_LEVEL) {
                        level[dst] = horizon + 1;
                        level_counts[horizon + 1]++;
                    }
                }
            }
        }
    }
}
