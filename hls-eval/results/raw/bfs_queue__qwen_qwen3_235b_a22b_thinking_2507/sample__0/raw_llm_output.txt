#include "bfs_queue.h"

void bfs(node_t nodes[N_NODES], edge_t edges[N_EDGES], node_index_t starting_node, level_t level[N_NODES], edge_index_t level_counts[N_LEVELS]) {
    // Initialize all levels to MAX_LEVEL (unvisited)
    for (node_index_t i = 0; i < N_NODES; i++) {
        level[i] = MAX_LEVEL;
    }
    level[starting_node] = 0;  // Starting node is at level 0

    // Initialize level counts to zero and set level 0 count
    for (edge_index_t i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }
    level_counts[0] = 1;

    // Circular queue implementation with size tracking
    node_index_t queue[N_NODES];
    int head = 0;
    int tail = 0;
    int queue_size = 0;

    // Enqueue starting node
    queue[tail] = starting_node;
    tail = (tail + 1) % N_NODES;
    queue_size = 1;

    // Process until queue is empty
    while (queue_size > 0) {
        // Dequeue node
        node_index_t current = queue[head];
        head = (head + 1) % N_NODES;
        queue_size--;

        level_t current_level = level[current];

        // Process all neighbors of current node
        edge_index_t begin = nodes[current].edge_begin;
        edge_index_t end = nodes[current].edge_end;
        for (edge_index_t e = begin; e < end; e++) {
            node_index_t neighbor = edges[e].dst;
            
            // Check if neighbor hasn't been visited
            if (level[neighbor] == MAX_LEVEL) {
                level_t next_level = current_level + 1;
                level[neighbor] = next_level;
                
                // Update level counts if within recorded range
                if (next_level < N_LEVELS) {
                    level_counts[next_level]++;
                }
                
                // Enqueue neighbor
                queue[tail] = neighbor;
                tail = (tail + 1) % N_NODES;
                queue_size++;
            }
        }
    }
}
