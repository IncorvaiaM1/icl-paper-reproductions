#include "bfs_queue.h"

#define QUEUE_SIZE (N_NODES + 1)

void bfs(node_t nodes[N_NODES], edge_t edges[N_EDGES], node_index_t starting_node, level_t level[N_NODES], edge_index_t level_counts[N_LEVELS]) {
    // Initialize level array to MAX_LEVEL
    for (node_index_t i = 0; i < N_NODES; i++) {
        level[i] = MAX_LEVEL;
    }

    // Initialize level_counts to zero
    for (edge_index_t i = 0; i < N_LEVELS; i++) {
        level_counts[i] = 0;
    }

    // Set the starting node
    level[starting_node] = 0;
    level_counts[0] = 1;

    // Create a queue with QUEUE_SIZE
    node_index_t queue[QUEUE_SIZE];
    int head = 0;
    int tail = 0;

    // Enqueue the starting node
    queue[tail] = starting_node;
    tail = (tail + 1) % QUEUE_SIZE;

    // While the queue is not empty
    while (head != tail) {
        // Dequeue the current node
        node_index_t current_node = queue[head];
        head = (head + 1) % QUEUE_SIZE;

        // Get the current level
        level_t current_level = level[current_node];
        level_t next_level = current_level + 1;

        // Process all edges of the current node
        edge_index_t start_edge = nodes[current_node].edge_begin;
        edge_index_t end_edge = nodes[current_node].edge_end;
        for (edge_index_t e = start_edge; e < end_edge; e++) {
            node_index_t neighbor = edges[e].dst;

            // If the neighbor has not been visited
            if (level[neighbor] == MAX_LEVEL) {
                // Update the level of the neighbor
                level[neighbor] = next_level;

                // Update the level_counts for the next level (if within bounds)
                if (next_level < N_LEVELS) {
                    level_counts[next_level]++;
                }

                // Enqueue the neighbor
                queue[tail] = neighbor;
                tail = (tail + 1) % QUEUE_SIZE;
            }
        }
    }
}
