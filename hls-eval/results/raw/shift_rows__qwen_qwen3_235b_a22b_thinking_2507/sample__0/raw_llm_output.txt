#include "shift_rows.h"

void ShiftRows(state_t *state) {
    // Row 1: shift left by 1 (row index 1)
    uint8_t temp1 = (*state)[1][0];
    (*state)[1][0] = (*state)[1][1];
    (*state)[1][1] = (*state)[1][2];
    (*state)[1][2] = (*state)[1][3];
    (*state)[1][3] = temp1;

    // Row 2: shift left by 2 (row index 2)
    uint8_t temp2_0 = (*state)[2][0];
    uint8_t temp2_1 = (*state)[2][1];
    (*state)[2][0] = (*state)[2][2];
    (*state)[2][1] = (*state)[2][3];
    (*state)[2][2] = temp2_0;
    (*state)[2][3] = temp2_1;

    // Row 3: shift left by 3 (row index 3) - equivalent to right shift by 1
    uint8_t temp3 = (*state)[3][3];
    (*state)[3][3] = (*state)[3][2];
    (*state)[3][2] = (*state)[3][1];
    (*state)[3][1] = (*state)[3][0];
    (*state)[3][0] = temp3;
}
