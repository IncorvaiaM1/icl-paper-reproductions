#include "seidel-2d.h"

void kernel_seidel_2d(double A[40][40]) {
    const int n = 40;
    const int tsteps = 20;

    double A_prev[40][40];

    // Copy the initial data to A_prev
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            A_prev[i][j] = A[i][j];
        }
    }

    for (int t = 0; t < tsteps; t++) {
        for (int i = 1; i < n - 1; i++) {
            for (int j = 1; j < n - 1; j++) {
                A[i][j] = (A_prev[i-1][j-1] + A_prev[i-1][j] + A_prev[i-1][j+1] +
                           A_prev[i][j-1] + A[i][j] + A_prev[i][j+1] +
                           A_prev[i+1][j-1] + A_prev[i+1][j] + A_prev[i+1][j+1]) / 9.0;
            }
        }

        // Update A_prev with the new values from A
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                A_prev[i][j] = A[i][j];
            }
        }
    }
}
