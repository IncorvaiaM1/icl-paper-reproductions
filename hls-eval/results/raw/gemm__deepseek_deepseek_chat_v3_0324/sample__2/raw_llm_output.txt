#include "gemm.h"

void kernel_gemm(
    double alpha,
    double beta,
    double C[20][25],
    double A[20][30],
    double B[30][25]) {

    // Temporary storage for the result of alpha * A * B
    double temp[20][25] = {0};

    // Compute the matrix multiplication: temp = alpha * A * B
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 25; j++) {
            double sum = 0;
            for (int k = 0; k < 30; k++) {
                sum += A[i][k] * B[k][j];
            }
            temp[i][j] = alpha * sum;
        }
    }

    // Compute the final result: C = temp + beta * C
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 25; j++) {
            C[i][j] = temp[i][j] + beta * C[i][j];
        }
    }
}
