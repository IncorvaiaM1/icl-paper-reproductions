#include "covariance.h"

void kernel_covariance(
    double float_n,
    double data[32][28],
    double cov[28][28],
    double mean[28])
{
    int i, j, k;
    double n = float_n;

    // Calculate mean of each attribute
    for (j = 0; j < 28; j++) {
        mean[j] = 0.0;
        for (i = 0; i < 32; i++) {
            mean[j] += data[i][j];
        }
        mean[j] /= n;
    }

    // Calculate covariance matrix
    for (i = 0; i < 28; i++) {
        for (j = i; j < 28; j++) {
            cov[i][j] = 0.0;
            for (k = 0; k < 32; k++) {
                cov[i][j] += (data[k][i] - mean[i]) * (data[k][j] - mean[j]);
            }
            cov[i][j] /= (n - 1.0);
            // Fill symmetric element
            if (i != j) {
                cov[j][i] = cov[i][j];
            }
        }
    }
}
