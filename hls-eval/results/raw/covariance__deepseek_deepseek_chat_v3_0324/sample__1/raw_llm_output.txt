#include "covariance.h"
#include <cmath>

void kernel_covariance(
    double float_n,
    double data[32][28],
    double cov[28][28],
    double mean[28])
{
    // Calculate the mean of each column
    for (int j = 0; j < 28; j++) {
        mean[j] = 0.0;
        for (int i = 0; i < 32; i++) {
            mean[j] += data[i][j];
        }
        mean[j] /= float_n;
    }

    // Center the data (subtract the mean from each column)
    for (int i = 0; i < 32; i++) {
        for (int j = 0; j < 28; j++) {
            data[i][j] -= mean[j];
        }
    }

    // Calculate the covariance matrix
    for (int j1 = 0; j1 < 28; j1++) {
        for (int j2 = j1; j2 < 28; j2++) {
            cov[j1][j2] = 0.0;
            for (int i = 0; i < 32; i++) {
                cov[j1][j2] += data[i][j1] * data[i][j2];
            }
            cov[j1][j2] /= (float_n - 1.0);
            // The matrix is symmetric
            if (j1 != j2) {
                cov[j2][j1] = cov[j1][j2];
            }
        }
    }
}
