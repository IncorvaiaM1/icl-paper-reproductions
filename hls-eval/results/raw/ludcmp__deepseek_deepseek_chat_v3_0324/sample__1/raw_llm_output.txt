#include "ludcmp.h"
#include <cmath>

void kernel_ludcmp(double A[40][40], double b[40], double x[40], double y[40]) {
    // LU Decomposition (in-place)
    for (int k = 0; k < 40; k++) {
        for (int i = k + 1; i < 40; i++) {
            A[i][k] = A[i][k] / A[k][k];
            for (int j = k + 1; j < 40; j++) {
                A[i][j] = A[i][j] - A[i][k] * A[k][j];
            }
        }
    }

    // Forward substitution: Ly = b
    for (int i = 0; i < 40; i++) {
        double sum = 0;
        for (int j = 0; j < i; j++) {
            sum += A[i][j] * y[j];
        }
        y[i] = (b[i] - sum) / A[i][i];
    }

    // Backward substitution: Ux = y
    for (int i = 39; i >= 0; i--) {
        double sum = 0;
        for (int j = i + 1; j < 40; j++) {
            sum += A[i][j] * x[j];
        }
        x[i] = (y[i] - sum) / A[i][i];
    }
}
