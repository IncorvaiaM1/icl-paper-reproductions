#include "bicg.h"

void kernel_bicg(double A[42][38], double s[38], double q[42], double p[38], double r[42]) {
    int i, j;

    // Compute q = A * p
    for (i = 0; i < 42; i++) {
        double sum = 0.0;
        for (j = 0; j < 38; j++) {
            sum += A[i][j] * p[j];
        }
        q[i] = sum;
    }

    // Compute s = A^T * r
    for (j = 0; j < 38; j++) {
        double sum = 0.0;
        for (i = 0; i < 42; i++) {
            sum += A[i][j] * r[i];
        }
        s[j] = sum;
    }
}
