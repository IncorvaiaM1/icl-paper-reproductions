#include "gemver.h"

void kernel_gemver(
    double alpha,
    double beta,
    double A[40][40],
    double u1[40],
    double v1[40],
    double u2[40],
    double v2[40],
    double w[40],
    double x[40],
    double y[40],
    double z[40]) {

    const int N = 40;

    // Stage 1: Matrix Update A = A + u1*v1^T + u2*v2^T
    MATRIX_UPDATE:
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            #pragma HLS PIPELINE II=1
            A[i][j] = A[i][j] + u1[i] * v1[j] + u2[i] * v2[j];
        }
    }

    // Stage 2: Vector Update x = beta*A^T*y + z
    VECTOR_UPDATE_X:
    for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE II=1
        double temp = 0.0;
        for (int j = 0; j < N; j++) {
            temp += A[j][i] * y[j];
        }
        x[i] = beta * temp + z[i];
    }

    // Stage 3: Final Vector Computation w = alpha*A*x
    VECTOR_UPDATE_W:
    for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE II=1
        double temp = 0.0;
        for (int j = 0; j < N; j++) {
            temp += A[i][j] * x[j];
        }
        w[i] = alpha * temp;
    }
}
