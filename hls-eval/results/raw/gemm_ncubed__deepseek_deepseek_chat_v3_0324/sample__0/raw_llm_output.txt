#include "gemm_ncubed.h"

void gemm(TYPE m1[N], TYPE m2[N], TYPE prod[N]) {
    #pragma HLS INTERFACE m_axi port=m1 bundle=gmem
    #pragma HLS INTERFACE m_axi port=m2 bundle=gmem
    #pragma HLS INTERFACE m_axi port=prod bundle=gmem
    #pragma HLS INTERFACE s_axilite port=return bundle=control

    // Initialize product matrix to zero
    for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE II=1
        prod[i] = 0;
    }

    // Perform matrix multiplication
    outer_loop: for (int i = 0; i < row_size; i++) {
        middle_loop: for (int j = 0; j < col_size; j++) {
            #pragma HLS PIPELINE II=1
            TYPE sum = 0;
            inner_loop: for (int k = 0; k < col_size; k++) {
                sum += m1[i * col_size + k] * m2[k * col_size + j];
            }
            prod[i * col_size + j] = sum;
        }
    }
}
