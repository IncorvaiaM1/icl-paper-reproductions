#include "nw_nw.h"

#define MATCH_SCORE 1
#define MISMATCH_SCORE -1
#define GAP_SCORE -1

#define M_IDX(i, j) ((i) * (BLEN+1) + (j))
#define PTR_IDX(i, j) ((i) * (BLEN+1) + (j))

void needwun(char SEQA[ALEN], char SEQB[BLEN], char alignedA[ALEN + BLEN], char alignedB[ALEN + BLEN], int M[(ALEN + 1) * (BLEN + 1)], char ptr[(ALEN + 1) * (BLEN + 1)]) {
    // Initialize aligned sequences to underscores
    for (int i = 0; i < ALEN+BLEN; i++) {
        alignedA[i] = '_';
        alignedB[i] = '_';
    }

    // Initialize boundary conditions for M and ptr matrices
    M[M_IDX(0,0)] = 0;
    ptr[PTR_IDX(0,0)] = '0';

    // Initialize first row (j from 1 to BLEN)
    for (int j = 1; j <= BLEN; j++) {
        M[M_IDX(0,j)] = j * GAP_SCORE;
        ptr[PTR_IDX(0,j)] = 'L';
    }

    // Initialize first column (i from 1 to ALEN)
    for (int i = 1; i <= ALEN; i++) {
        M[M_IDX(i,0)] = i * GAP_SCORE;
        ptr[PTR_IDX(i,0)] = 'U';
    }

    // Fill the similarity matrix
    for (int i = 1; i <= ALEN; i++) {
        for (int j = 1; j <= BLEN; j++) {
            int diag = M[M_IDX(i-1, j-1)] + (SEQA[i-1] == SEQB[j-1] ? MATCH_SCORE : MISMATCH_SCORE);
            int up = M[M_IDX(i-1, j)] + GAP_SCORE;
            int left = M[M_IDX(i, j-1)] + GAP_SCORE;

            int max_val = diag;
            char direction = 'D';

            if (up > max_val) {
                max_val = up;
                direction = 'U';
            }
            if (left > max_val) {
                max_val = left;
                direction = 'L';
            }

            M[M_IDX(i,j)] = max_val;
            ptr[PTR_IDX(i,j)] = direction;
        }
    }

    // Traceback to generate aligned sequences
    int i = ALEN;
    int j = BLEN;
    int pos = ALEN+BLEN-1;

    while (i > 0 || j > 0) {
        char p = ptr[PTR_IDX(i, j)];
        switch (p) {
            case 'D':
                alignedA[pos] = SEQA[i-1];
                alignedB[pos] = SEQB[j-1];
                i--;
                j--;
                break;
            case 'U':
                alignedA[pos] = SEQA[i-1];
                alignedB[pos] = '_';
                i--;
                break;
            case 'L':
                alignedA[pos] = '_';
                alignedB[pos] = SEQB[j-1];
                j--;
                break;
            default:
                i = 0;
                j = 0;
                break;
        }
        pos--;
    }

    // Shift alignment to beginning of arrays
    int len = ALEN+BLEN - (pos+1);
    for (int k = 0; k < len; k++) {
        alignedA[k] = alignedA[pos+1+k];
        alignedB[k] = alignedB[pos+1+k];
    }
}
