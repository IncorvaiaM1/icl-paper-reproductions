#include "lu.h"

void kernel_lu(double A[40][40]) {
    double L[40][40] = {0};
    double U[40][40] = {0};
    
    // LU Decomposition without pivoting
    for (int i = 0; i < 40; i++) {
        // Compute U matrix elements
        for (int j = i; j < 40; j++) {
            double sum = 0.0;
            for (int k = 0; k < i; k++) {
                sum += L[i][k] * U[k][j];
            }
            U[i][j] = A[i][j] - sum;
        }
        
        // Compute L matrix elements
        for (int j = i; j < 40; j++) {
            if (i == j) {
                L[i][i] = 1.0;
            } else {
                double sum = 0.0;
                for (int k = 0; k < i; k++) {
                    sum += L[j][k] * U[k][i];
                }
                L[j][i] = (A[j][i] - sum) / U[i][i];
            }
        }
    }
    
    // Copy results back to input matrix A as L and U
    // Lower part (including diagonal) is L, upper part is U
    for (int i = 0; i < 40; i++) {
        for (int j = 0; j < 40; j++) {
            if (i > j) {
                A[i][j] = L[i][j];
            } else {
                A[i][j] = U[i][j];
            }
        }
    }
}
