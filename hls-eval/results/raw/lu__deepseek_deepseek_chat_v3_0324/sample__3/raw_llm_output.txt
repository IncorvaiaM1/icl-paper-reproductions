#include "lu.h"

void kernel_lu(double A[40][40]) {
    #pragma HLS INTERFACE m_axi port=A depth=1600
    #pragma HLS INTERFACE s_axilite port=return

    double L[40][40] = {0};
    double U[40][40] = {0};

    // LU decomposition without pivoting
    for (int i = 0; i < 40; i++) {
        #pragma HLS PIPELINE II=1
        // Compute U elements for row i
        for (int j = i; j < 40; j++) {
            #pragma HLS PIPELINE II=1
            double sum = 0.0;
            for (int k = 0; k < i; k++) {
                sum += L[i][k] * U[k][j];
            }
            U[i][j] = A[i][j] - sum;
        }

        // Compute L elements for column i
        for (int j = i; j < 40; j++) {
            #pragma HLS PIPELINE II=1
            if (i == j) {
                L[i][i] = 1.0;
            } else {
                double sum = 0.0;
                for (int k = 0; k < i; k++) {
                    sum += L[j][k] * U[k][i];
                }
                L[j][i] = (A[j][i] - sum) / U[i][i];
            }
        }
    }

    // Copy results back to A (L in lower part, U in upper part)
    for (int i = 0; i < 40; i++) {
        #pragma HLS PIPELINE II=1
        for (int j = 0; j < 40; j++) {
            if (i > j) {
                A[i][j] = L[i][j];
            } else {
                A[i][j] = U[i][j];
            }
        }
    }
}
