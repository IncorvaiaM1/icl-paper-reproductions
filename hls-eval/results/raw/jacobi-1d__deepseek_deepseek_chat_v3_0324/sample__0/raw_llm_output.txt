#include "jacobi-1d.h"

void kernel_jacobi_1d(double A[30], double B[30]) {
    const int n = 30;
    const int tsteps = 20;
    double temp[30];

    for (int t = 0; t < tsteps; t++) {
        // Compute stencil operation
        for (int i = 1; i < n - 1; i++) {
            temp[i] = (A[i-1] + A[i] + A[i+1]) / 3.0;
        }

        // Handle boundary conditions (unchanged)
        temp[0] = A[0];
        temp[n-1] = A[n-1];

        // Copy results back to A for next iteration
        for (int i = 0; i < n; i++) {
            A[i] = temp[i];
        }
    }

    // Copy final result to output array B
    for (int i = 0; i < n; i++) {
        B[i] = A[i];
    }
}
