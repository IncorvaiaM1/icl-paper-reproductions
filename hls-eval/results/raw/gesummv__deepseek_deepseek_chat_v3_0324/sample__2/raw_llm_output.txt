#include "gesummv.h"

void kernel_gesummv(
    double alpha,
    double beta,
    double A[30][30],
    double B[30][30],
    double tmp[30],
    double x[30],
    double y[30])
{
    const int N = 30;

    // Compute tmp = A * x
    for (int i = 0; i < N; i++) {
        tmp[i] = 0.0;
        for (int j = 0; j < N; j++) {
            tmp[i] += A[i][j] * x[j];
        }
    }

    // Compute y = alpha * tmp (which is alpha * A * x)
    for (int i = 0; i < N; i++) {
        y[i] = alpha * tmp[i];
    }

    // Compute tmp = B * x
    for (int i = 0; i < N; i++) {
        tmp[i] = 0.0;
        for (int j = 0; j < N; j++) {
            tmp[i] += B[i][j] * x[j];
        }
    }

    // Compute y = y + beta * tmp (which is alpha * A * x + beta * B * x)
    for (int i = 0; i < N; i++) {
        y[i] += beta * tmp[i];
    }
}
