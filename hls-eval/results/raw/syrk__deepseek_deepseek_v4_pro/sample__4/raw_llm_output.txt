#include "syrk.h"

void kernel_syrk(double alpha, double beta, double C[30][30], double A[30][20]) {
    const int N = 30;
    const int M = 20;

    // SYRK: C = alpha * A * A^T + beta * C
    // Only the lower triangular part of C is updated (UPLO = 'L')
    for (int i = 0; i < N; i++) {
        for (int j = 0; j <= i; j++) {
            double sum = 0.0;
            // Dot product of row i and row j of A
            for (int k = 0; k < M; k++) {
                sum += A[i][k] * A[j][k];
            }
            C[i][j] = beta * C[i][j] + alpha * sum;
        }
    }
}
