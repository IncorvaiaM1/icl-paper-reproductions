#include "syrk.h"

void kernel_syrk(double alpha, double beta, double C[30][30], double A[30][20]) {
    // Compute C = alpha * A * A^T + beta * C
    // Since C is lower triangular (UPLO = 'L'), we only compute and update
    // the lower triangular part of C (i >= j).

    // First, scale C by beta
    for (int i = 0; i < 30; i++) {
        for (int j = 0; j <= i; j++) {
            C[i][j] = beta * C[i][j];
        }
    }

    // Then add alpha * A * A^T to C
    // For lower triangular part: C[i][j] += alpha * sum_{k=0}^{19} A[i][k] * A[j][k]
    for (int i = 0; i < 30; i++) {
        for (int j = 0; j <= i; j++) {
            double sum = 0.0;
            for (int k = 0; k < 20; k++) {
                sum += A[i][k] * A[j][k];
            }
            C[i][j] += alpha * sum;
        }
    }
}
