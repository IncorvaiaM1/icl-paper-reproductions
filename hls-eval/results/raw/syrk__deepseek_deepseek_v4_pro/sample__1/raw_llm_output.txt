#include "syrk.h"

void kernel_syrk(double alpha, double beta, double C[30][30], double A[30][20]) {
    // UPLO = 'L' : lower triangular part of C is stored and updated.
    // TRANS = 'N' : A is not transposed, compute A * A^T.
    for (int i = 0; i < 30; i++) {
        for (int j = 0; j <= i; j++) {
            double sum = 0.0;
            for (int k = 0; k < 20; k++) {
                sum += A[i][k] * A[j][k];
            }
            C[i][j] = alpha * sum + beta * C[i][j];
        }
    }
}
