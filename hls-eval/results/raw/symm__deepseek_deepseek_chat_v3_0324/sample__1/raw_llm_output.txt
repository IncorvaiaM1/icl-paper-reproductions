#include "symm.h"

void kernel_symm(
    double alpha,
    double beta,
    double C[20][30],
    double A[20][20],
    double B[20][30]) {

    double temp_C[20][30];

    // Initialize temp_C with beta * B * C
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            temp_C[i][j] = beta * B[i][j] * C[i][j];
        }
    }

    // Compute alpha * A * B and accumulate to temp_C
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            double sum = 0.0;
            // Since A is symmetric and stored as lower triangular,
            // we only need to consider elements where k <= i
            for (int k = 0; k <= i; k++) {
                sum += A[i][k] * B[k][j];
            }
            // For upper triangular elements (k > i), use symmetry: A[k][i] = A[i][k]
            for (int k = i + 1; k < 20; k++) {
                sum += A[k][i] * B[k][j];
            }
            temp_C[i][j] += alpha * sum;
        }
    }

    // Copy results back to C
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < 30; j++) {
            C[i][j] = temp_C[i][j];
        }
    }
}
