void merge(TYPE a[SIZE], TYPE temp[SIZE], int start, int mid, int end) {
    merge_label1: for (int i = start; i < mid; i++) {
        temp[i] = a[i];
    }
    
    merge_label2: for (int i = mid; i <= end; i++) {
        temp[i] = a[i];
    }
    
    int i = start, j = mid;
    merge_label3: for (int k = start; k <= end; k++) {
        if (i < mid && (j > end || temp[i] <= temp[j])) {
            a[k] = temp[i];
            i++;
        } else {
            a[k] = temp[j];
            j++;
        }
    }
}

void ms_mergesort(TYPE a[SIZE]) {
    TYPE temp[SIZE];
    
    for (int width = 1; width < SIZE; width *= 2) {
        for (int i = 0; i < SIZE; i += 2 * width) {
            int start = i;
            int mid = i + width;
            if (mid >= SIZE) {
                break;
            }
            int end = i + 2 * width - 1;
            if (end >= SIZE) {
                end = SIZE - 1;
            }
            
            merge(a, temp, start, mid, end);
        }
    }
}
