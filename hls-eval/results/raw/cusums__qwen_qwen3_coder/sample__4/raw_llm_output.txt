#include "cusums.h"

void CumulativeSums(int *res_sup, int *res_inf, int epsilon[N]) {
    int S = 0;
    int sup = 0;
    int inf = 0;

    // Initialize sup and inf based on initial value of S
    sup = S;
    inf = S;

    // Iterate through the epsilon array
    for (int k = 0; k < N; k++) {
        // Update cumulative sum S
        if (epsilon[k] == 1) {
            S += 1;
        } else {
            S -= 1;
        }

        // Update maximum and minimum values
        if (S > sup) {
            sup = S;
        }
        if (S < inf) {
            inf = S;
        }
    }

    // Store results
    *res_sup = sup;
    *res_inf = inf;
}
