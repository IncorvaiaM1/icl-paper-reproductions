#include "covariance.h"

void kernel_covariance(
    double float_n,
    double data[32][28],
    double cov[28][28],
    double mean[28]) {

    int i, j, k;
    double temp[32][28];

    // Step 1: Compute mean for each attribute
    for (j = 0; j < 28; j++) {
        mean[j] = 0.0;
        for (k = 0; k < 32; k++) {
            mean[j] += data[k][j];
        }
        mean[j] /= float_n;
    }

    // Step 2: Subtract mean from each data point
    for (k = 0; k < 32; k++) {
        for (j = 0; j < 28; j++) {
            temp[k][j] = data[k][j] - mean[j];
        }
    }

    // Step 3: Compute covariance matrix
    for (i = 0; i < 28; i++) {
        for (j = 0; j < 28; j++) {
            cov[i][j] = 0.0;
            for (k = 0; k < 32; k++) {
                cov[i][j] += temp[k][i] * temp[k][j];
            }
            cov[i][j] /= (float_n - 1);
        }
    }
}
