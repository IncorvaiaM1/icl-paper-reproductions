#include "shift_rows.h"

void ShiftRows(state_t *state) {
#pragma HLS INLINE off
    // Row 0: no shift
    // Row 1: left shift by 1
    uint8_t tmp1 = (*state)[1][0];
    (*state)[1][0] = (*state)[1][1];
    (*state)[1][1] = (*state)[1][2];
    (*state)[1][2] = (*state)[1][3];
    (*state)[1][3] = tmp1;

    // Row 2: left shift by 2
    uint8_t tmp2_0 = (*state)[2][0];
    uint8_t tmp2_1 = (*state)[2][1];
    (*state)[2][0] = (*state)[2][2];
    (*state)[2][1] = (*state)[2][3];
    (*state)[2][2] = tmp2_0;
    (*state)[2][3] = tmp2_1;

    // Row 3: left shift by 3
    uint8_t tmp3_0 = (*state)[3][0];
    (*state)[3][0] = (*state)[3][3];
    (*state)[3][3] = (*state)[3][2];
    (*state)[3][2] = (*state)[3][1];
    (*state)[3][1] = tmp3_0;
}
