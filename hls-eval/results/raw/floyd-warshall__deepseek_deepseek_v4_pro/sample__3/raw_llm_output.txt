#include "floyd-warshall.h"

void kernel_floyd_warshall(int path[60][60]) {
    for (int k = 0; k < 60; ++k) {
        for (int i = 0; i < 60; ++i) {
            for (int j = 0; j < 60; ++j) {
                int via = path[i][k] + path[k][j];
                if (via < path[i][j]) {
                    path[i][j] = via;
                }
            }
        }
    }
}
