#include "kmp_kmp.h"

void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    int32_t k = 0;
    int32_t q;

    kmpNext[0] = 0;

    for (q = 1; q < PATTERN_SIZE; q++) {
        while (k > 0 && pattern[k] != pattern[q]) {
            k = kmpNext[k - 1];
        }
        if (pattern[k] == pattern[q]) {
            k++;
        }
        kmpNext[q] = k;
    }
}

int kmp(
    char pattern[PATTERN_SIZE],
    char input[STRING_SIZE],
    int32_t kmpNext[PATTERN_SIZE],
    int32_t n_matches[1]) {
    
    int32_t k = 0;
    int32_t q;
    int32_t i;
    n_matches[0] = 0;

    CPF(pattern, kmpNext);

    for (i = 0; i < STRING_SIZE; i++) {
        while (k > 0 && pattern[k] != input[i]) {
            k = kmpNext[k - 1];
        }
        if (pattern[k] == input[i]) {
            k++;
        }
        if (k == PATTERN_SIZE) {
            n_matches[0]++;
            k = kmpNext[k - 1];
        }
    }

    return n_matches[0];
}
