#include "kmp_kmp.h"

void CPF(char pattern[PATTERN_SIZE], int32_t kmpNext[PATTERN_SIZE]) {
    int32_t i, j;

    i = 0;
    j = -1;
    kmpNext[0] = -1;
    while (i < PATTERN_SIZE - 1) {
        if (j == -1 || pattern[i] == pattern[j]) {
            i++;
            j++;
            kmpNext[i] = j;
        } else {
            j = kmpNext[j];
        }
    }
}

int kmp(char pattern[PATTERN_SIZE], char input[STRING_SIZE],
        int32_t kmpNext[PATTERN_SIZE], int32_t n_matches[1]) {
#pragma HLS INTERFACE m_axi port = pattern bundle = gmem0
#pragma HLS INTERFACE m_axi port = input bundle = gmem1
#pragma HLS INTERFACE m_axi port = kmpNext bundle = gmem2
#pragma HLS INTERFACE m_axi port = n_matches bundle = gmem3

#pragma HLS INTERFACE s_axilite port = pattern bundle = control
#pragma HLS INTERFACE s_axilite port = input bundle = control
#pragma HLS INTERFACE s_axilite port = kmpNext bundle = control
#pragma HLS INTERFACE s_axilite port = n_matches bundle = control
#pragma HLS INTERFACE s_axilite port = return bundle = control

    int32_t i, q;

    // Preprocess the pattern to build the kmpNext table
    CPF(pattern, kmpNext);

    n_matches[0] = 0;
    q = 0;

    // Search for the pattern in the input string
    for (i = 0; i < STRING_SIZE; i++) {
#pragma HLS LOOP_TRIPCOUNT min = 1 max = STRING_SIZE
        while (q > 0 && pattern[q] != input[i]) {
            q = kmpNext[q];
        }
        if (pattern[q] == input[i]) {
            q++;
        }
        if (q == PATTERN_SIZE) {
            n_matches[0]++;
            q = kmpNext[q - 1];
        }
    }

    return 0;
}
