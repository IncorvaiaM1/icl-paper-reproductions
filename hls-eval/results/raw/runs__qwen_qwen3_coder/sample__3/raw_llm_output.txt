#include "runs.h"

void Runs(int *res_S, int *res_V, int epsilon[N]) {
    int S = 0;
    int V = 0;

    if (N > 0) {
        // Initialize run count with the first element
        V = 1;

        // Count the first element if it's 1
        if (epsilon[0] == 1) {
            S = 1;
        } else {
            S = 0;
        }

        // Iterate through the rest of the array
        for (int i = 1; i < N; i++) {
            // Increment S if current element is 1
            if (epsilon[i] == 1) {
                S++;
            }

            // Check for transition (run change)
            if (epsilon[i] != epsilon[i - 1]) {
                V++;
            }
        }
    }

    *res_S = S;
    *res_V = V;
}
