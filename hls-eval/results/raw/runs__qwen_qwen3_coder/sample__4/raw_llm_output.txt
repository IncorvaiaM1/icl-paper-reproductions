#include "runs.h"

void Runs(int *res_S, int *res_V, int epsilon[N]) {
    int S = 0;
    int V = 0;

    if (N > 0) {
        V = 1; // At least one run if array is non-empty
    }

    for (int i = 0; i < N; i++) {
        if (epsilon[i] == 1) {
            S++;
        }

        if (i > 0 && epsilon[i] != epsilon[i - 1]) {
            V++;
        }
    }

    *res_S = S;
    *res_V = V;
}
