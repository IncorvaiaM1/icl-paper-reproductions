#include "trisolv.h"

void kernel_trisolv(double L[40][40], double x[40], double b[40]) {
    for (int i = 0; i < 40; i++) {
        double sum = b[i];
        for (int j = 0; j < i; j++) {
            sum -= L[i][j] * x[j];
        }
        x[i] = sum / L[i][i];
    }
}
