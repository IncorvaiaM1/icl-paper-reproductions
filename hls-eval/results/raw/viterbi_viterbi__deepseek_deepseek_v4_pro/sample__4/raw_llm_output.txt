#include "viterbi_viterbi.h"

int viterbi(
    tok_t obs[N_OBS],
    prob_t init[N_STATES],
    prob_t transition[N_STATES * N_STATES],
    prob_t emission[N_STATES * N_TOKENS],
    state_t path[N_OBS])
{
    prob_t llike[N_OBS][N_STATES];
    step_t t, i, j;
    prob_t min_val, temp;
    state_t min_state;

    // L_init: initialization with first observation
    for (i = 0; i < N_STATES; i++) {
        llike[0][i] = init[i] + emission[i][obs[0]];
    }

    // L_timestep: iterate over remaining observations
    for (t = 1; t < N_OBS; t++) {
        // L_curr_state: for each current state
        for (j = 0; j < N_STATES; j++) {
            // L_prev_state: find most likely previous state
            min_val = llike[t-1][0] + transition[0 * N_STATES + j] + emission[j][obs[t]];
            min_state = 0;
            for (i = 1; i < N_STATES; i++) {
                temp = llike[t-1][i] + transition[i * N_STATES + j] + emission[j][obs[t]];
                if (temp < min_val) {
                    min_val = temp;
                    min_state = i;
                }
            }
            llike[t][j] = min_val;
        }
    }

    // L_end: identify end state with maximum probability (minimum -log prob)
    min_val = llike[N_OBS-1][0];
    path[N_OBS-1] = 0;
    for (i = 1; i < N_STATES; i++) {
        if (llike[N_OBS-1][i] < min_val) {
            min_val = llike[N_OBS-1][i];
            path[N_OBS-1] = i;
        }
    }

    // L_backtrack: recover most likely state sequence
    for (t = N_OBS-1; t > 0; t--) {
        j = path[t];
        min_val = llike[t-1][0] + transition[0 * N_STATES + j];
        min_state = 0;
        for (i = 1; i < N_STATES; i++) {
            temp = llike[t-1][i] + transition[i * N_STATES + j];
            if (temp < min_val) {
                min_val = temp;
                min_state = i;
            }
        }
        path[t-1] = min_state;
    }

    return 0;
}
