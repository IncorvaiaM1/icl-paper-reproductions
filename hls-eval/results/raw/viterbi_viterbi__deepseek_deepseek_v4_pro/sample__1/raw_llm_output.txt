#include "viterbi_viterbi.h"

int viterbi(
    tok_t obs[N_OBS],
    prob_t init[N_STATES],
    prob_t transition[N_STATES * N_STATES],
    prob_t emission[N_STATES * N_TOKENS],
    state_t path[N_OBS])
{
    // Check for invalid observations
    for (int t = 0; t < N_OBS; t++) {
        if (obs[t] >= N_TOKENS) {
            return -1;
        }
    }

    prob_t llike[N_OBS][N_STATES];
    state_t backptr[N_OBS][N_STATES];

    // Initialization (L_init)
    for (int i = 0; i < N_STATES; i++) {
        llike[0][i] = init[i] + emission[i * N_TOKENS + obs[0]];
        backptr[0][i] = 0;
    }

    // Iterative computation (L_timestep)
    for (step_t t = 1; t < N_OBS; t++) {
        // L_curr_state
        for (int j = 0; j < N_STATES; j++) {
            prob_t min_prob = 1e100;
            state_t best_prev = 0;
            // L_prev_state
            for (int i = 0; i < N_STATES; i++) {
                prob_t prob = llike[t-1][i] + transition[i * N_STATES + j] + emission[j * N_TOKENS + obs[t]];
                if (prob < min_prob) {
                    min_prob = prob;
                    best_prev = i;
                }
            }
            llike[t][j] = min_prob;
            backptr[t][j] = best_prev;
        }
    }

    // L_end: find end state with minimum probability
    prob_t min_final = llike[N_OBS-1][0];
    state_t best_state = 0;
    for (int i = 1; i < N_STATES; i++) {
        if (llike[N_OBS-1][i] < min_final) {
            min_final = llike[N_OBS-1][i];
            best_state = i;
        }
    }

    // L_backtrack: recover path
    path[N_OBS-1] = best_state;
    for (step_t t = N_OBS-2; t >= 0; t--) {
        best_state = backptr[t+1][best_state];
        path[t] = best_state;
    }

    return 0;
}
