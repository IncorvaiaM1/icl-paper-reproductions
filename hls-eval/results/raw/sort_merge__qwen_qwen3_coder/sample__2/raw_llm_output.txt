static TYPE temp[SIZE];

void merge(TYPE a[SIZE], int start, int m, int stop) {
    int i, j, k;

merge_label1:
    for (i = start; i <= m; i++) {
        temp[i] = a[i];
    }

merge_label2:
    for (j = m + 1; j <= stop; j++) {
        temp[j] = a[j];
    }

    i = start;
    j = m + 1;
    k = start;

merge_label3:
    while (k <= stop) {
        if (i <= m && (j > stop || temp[i] <= temp[j])) {
            a[k] = temp[i];
            i++;
        } else {
            a[k] = temp[j];
            j++;
        }
        k++;
    }
}

void ms_mergesort(TYPE a[SIZE]) {
    int l, m, n;

    for (l = 1; l < SIZE; l = 2 * l) {
        for (m = 0; m < SIZE - 1; m += 2 * l) {
            n = m + 2 * l - 1;
            if (n >= SIZE) {
                n = SIZE - 1;
            }
            merge(a, m, m + l - 1, n);
        }
    }
}
