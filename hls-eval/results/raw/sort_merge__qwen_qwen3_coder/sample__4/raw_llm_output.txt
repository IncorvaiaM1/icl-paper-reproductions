void merge(TYPE a[SIZE], int start, int m, int stop) {

merge_label1:

merge_label2:

void ms_mergesort(TYPE a[SIZE]) {
    int l, m, h, i;

    l = 1;
    while (l < SIZE) {
        m = l - 1;
        while (m < SIZE - 1) {
            h = (m + l < SIZE - 1) ? m + l : SIZE - 1;
            merge(a, m - l + 1, m, h);
            m = m + l * 2;
        }
        l = l * 2;
    }
