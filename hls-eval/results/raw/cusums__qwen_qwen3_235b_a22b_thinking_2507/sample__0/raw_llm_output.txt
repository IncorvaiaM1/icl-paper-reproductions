#include "cusums.h"

void CumulativeSums(int *res_sup, int *res_inf, int epsilon[N]) {
    int S = 0;
    int sup = 0;
    int inf = 0;

    for (int k = 0; k < N; k++) {
        #pragma HLS PIPELINE
        S += (2 * epsilon[k] - 1);
        sup = (S > sup) ? S : sup;
        inf = (S < inf) ? S : inf;
    }

    *res_sup = sup;
    *res_inf = inf;
}
