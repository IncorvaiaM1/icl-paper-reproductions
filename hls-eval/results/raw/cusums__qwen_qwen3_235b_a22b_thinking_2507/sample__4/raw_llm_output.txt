#include "cusums.h"

void CumulativeSums(int *res_sup, int *res_inf, int epsilon[N]) {
    int S = 0;
    int sup = 0;
    int inf = 0;

    for (int i = 0; i < N; i++) {
        #pragma HLS PIPELINE
        if (epsilon[i] == 1) {
            S = S + 1;
        } else {
            S = S - 1;
        }
        if (S > sup) {
            sup = S;
        }
        if (S < inf) {
            inf = S;
        }
    }

    *res_sup = sup;
    *res_inf = inf;
}
