#include "lu.h"

void kernel_lu(double A[40][40]) {
    for (int i = 0; i < 40; i++) {
        for (int j = i; j < 40; j++) {
            double sum = 0.0;
            for (int k = 0; k < i; k++) {
                sum += A[i][k] * A[k][j];
            }
            A[i][j] -= sum;
        }
        
        for (int j = i + 1; j < 40; j++) {
            double sum = 0.0;
            for (int k = 0; k < i; k++) {
                sum += A[j][k] * A[k][i];
            }
            A[j][i] = (A[j][i] - sum) / A[i][i];
        }
    }
}
