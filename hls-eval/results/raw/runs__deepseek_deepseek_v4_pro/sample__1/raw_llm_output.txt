#include "runs.h"

void Runs(int *res_S, int *res_V, int epsilon[N]) {
#pragma HLS INTERFACE mode=m_axi port=epsilon offset=slave bundle=gmem
#pragma HLS INTERFACE mode=s_axilite port=res_S bundle=control
#pragma HLS INTERFACE mode=s_axilite port=res_V bundle=control
#pragma HLS INTERFACE mode=s_axilite port=return bundle=control

    int ones = 0;
    int runs = 1;
    int prev = epsilon[0];

    if (prev == 1) {
        ones += 1;
    }

    for (int i = 1; i < N; i++) {
#pragma HLS PIPELINE II=1
        int cur = epsilon[i];
        if (cur == 1) {
            ones += 1;
        }
        if (cur != prev) {
            runs += 1;
        }
        prev = cur;
    }

    *res_S = ones;
    *res_V = runs;
}
