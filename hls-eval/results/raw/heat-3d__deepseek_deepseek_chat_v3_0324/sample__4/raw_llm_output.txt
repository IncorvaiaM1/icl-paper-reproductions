#include "heat-3d.h"

void kernel_heat_3d(double A[10][10][10], double B[10][10][10]) {
    const int n = 10;
    const int tsteps = 20;
    const double alpha = 0.125;

    // Time steps loop
    for (int t = 0; t < tsteps; t++) {
        // Update grid points
        for (int i = 1; i < n-1; i++) {
            for (int j = 1; j < n-1; j++) {
                for (int k = 1; k < n-1; k++) {
                    // X-direction contribution
                    double x_diff = A[i+1][j][k] - 2.0 * A[i][j][k] + A[i-1][j][k];
                    // Y-direction contribution
                    double y_diff = A[i][j+1][k] - 2.0 * A[i][j][k] + A[i][j-1][k];
                    // Z-direction contribution
                    double z_diff = A[i][j][k+1] - 2.0 * A[i][j][k] + A[i][j][k-1];
                    
                    // Update equation
                    B[i][j][k] = A[i][j][k] + alpha * (x_diff + y_diff + z_diff);
                }
            }
        }

        // Swap arrays for next iteration
        for (int i = 1; i < n-1; i++) {
            for (int j = 1; j < n-1; j++) {
                for (int k = 1; k < n-1; k++) {
                    A[i][j][k] = B[i][j][k];
                }
            }
        }
    }
}
