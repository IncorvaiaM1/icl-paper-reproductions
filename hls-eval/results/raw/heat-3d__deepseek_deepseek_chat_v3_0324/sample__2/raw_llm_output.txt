#include "heat-3d.h"

void kernel_heat_3d(double A[10][10][10], double B[10][10][10]) {
    const int tsteps = 20;
    const int n = 10;
    
    // Temporary arrays to hold data between time steps
    double temp1[10][10][10];
    double temp2[10][10][10];
    
    // Initialize temp1 with input data
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            for (int k = 0; k < n; k++) {
                temp1[i][j][k] = A[i][j][k];
            }
        }
    }
    
    // Time steps loop
    for (int t = 0; t < tsteps; t++) {
        // Update grid points
        for (int i = 1; i < n-1; i++) {
            for (int j = 1; j < n-1; j++) {
                for (int k = 1; k < n-1; k++) {
                    // X direction contribution
                    double x_contrib = 0.125 * (temp1[i+1][j][k] - 2.0 * temp1[i][j][k] + temp1[i-1][j][k]);
                    // Y direction contribution
                    double y_contrib = 0.125 * (temp1[i][j+1][k] - 2.0 * temp1[i][j][k] + temp1[i][j-1][k]);
                    // Z direction contribution
                    double z_contrib = 0.125 * (temp1[i][j][k+1] - 2.0 * temp1[i][j][k] + temp1[i][j][k-1]);
                    
                    // Update value
                    temp2[i][j][k] = temp1[i][j][k] + x_contrib + y_contrib + z_contrib;
                }
            }
        }
        
        // Copy boundary conditions (unchanged)
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (i == 0 || i == n-1 || j == 0 || j == n-1 || k == 0 || k == n-1) {
                        temp2[i][j][k] = temp1[i][j][k];
                    }
                }
            }
        }
        
        // Swap arrays for next iteration
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    temp1[i][j][k] = temp2[i][j][k];
                }
            }
        }
    }
    
    // Copy final result to output
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            for (int k = 0; k < n; k++) {
                B[i][j][k] = temp1[i][j][k];
            }
        }
    }
}
