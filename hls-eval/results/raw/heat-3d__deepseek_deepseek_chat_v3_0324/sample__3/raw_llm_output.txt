#include "heat-3d.h"

void kernel_heat_3d(double A[10][10][10], double B[10][10][10]) {
    const int TSTEPS = 20;
    const int N = 10;
    
    // Temporary arrays to hold the data between time steps
    double temp1[N][N][N];
    double temp2[N][N][N];
    
    // Initialize temp1 with input data
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            for (int k = 0; k < N; k++) {
                temp1[i][j][k] = A[i][j][k];
                temp2[i][j][k] = 0.0;
            }
        }
    }
    
    // Time iteration loop
    for (int t = 0; t < TSTEPS; t++) {
        // Update each grid point
        for (int i = 1; i < N-1; i++) {
            for (int j = 1; j < N-1; j++) {
                for (int k = 1; k < N-1; k++) {
                    temp2[i][j][k] = temp1[i][j][k] +
                        0.125 * (temp1[i+1][j][k] - 2 * temp1[i][j][k] + temp1[i-1][j][k]) +
                        0.125 * (temp1[i][j+1][k] - 2 * temp1[i][j][k] + temp1[i][j-1][k]) +
                        0.125 * (temp1[i][j][k+1] - 2 * temp1[i][j][k] + temp1[i][j][k-1]);
                }
            }
        }
        
        // Swap buffers for next iteration
        for (int i = 1; i < N-1; i++) {
            for (int j = 1; j < N-1; j++) {
                for (int k = 1; k < N-1; k++) {
                    temp1[i][j][k] = temp2[i][j][k];
                }
            }
        }
    }
    
    // Copy final result to output array
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            for (int k = 0; k < N; k++) {
                B[i][j][k] = temp1[i][j][k];
            }
        }
    }
}
