#include "block.h"

void BlockFrequency(double *result, int epsilon[M * N]) {
    double sum = 0.0;
    
    for (int i = 0; i < M; i++) {
        int blockSum = 0;
        
        // Calculate sum of elements in current row
        for (int j = 0; j < N; j++) {
            blockSum += epsilon[i * N + j];
        }
        
        // Calculate proportion of 1s in the row
        double pi = (double)blockSum / (double)M;
        
        // Compute deviation from 0.5 and square it
        double v = pi - 0.5;
        sum += v * v;
    }
    
    *result = sum;
}
