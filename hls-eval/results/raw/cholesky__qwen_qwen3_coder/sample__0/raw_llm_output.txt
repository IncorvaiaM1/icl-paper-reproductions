#include "cholesky.h"

void kernel_cholesky(double A[40][40]) {
    int i, j, k;
    double sum;

    loop_i:
    for (i = 0; i < 40; i++) {
        loop_j:
        for (j = 0; j <= i; j++) {
            sum = 0.0;

            loop_k:
            for (k = 0; k < j; k++) {
                sum += A[i][k] * A[j][k];
            }

            if (i == j) {
                A[i][j] = sqrt(A[i][i] - sum);
            } else {
                A[i][j] = (A[i][j] - sum) / A[j][j];
            }
        }
    }
}
