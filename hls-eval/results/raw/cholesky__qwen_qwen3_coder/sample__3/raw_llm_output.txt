#include "cholesky.h"

void kernel_cholesky(double A[40][40]) {
    for (int i = 0; i < 40; i++) {
        // Compute diagonal element A[i][i]
        double sum_diag = 0.0;
        for (int k = 0; k < i; k++) {
            sum_diag += A[i][k] * A[i][k];
        }
        A[i][i] = sqrt(A[i][i] - sum_diag);

        // Compute elements below diagonal in column i
        for (int j = i + 1; j < 40; j++) {
            double sum_off_diag = 0.0;
            for (int k = 0; k < i; k++) {
                sum_off_diag += A[j][k] * A[i][k];
            }
            A[j][i] = (A[j][i] - sum_off_diag) / A[i][i];
        }

        // Zero out the upper triangular part for this column
        for (int j = 0; j < i; j++) {
            A[j][i] = 0.0;
        }
    }
}
