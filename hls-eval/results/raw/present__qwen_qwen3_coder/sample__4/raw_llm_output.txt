#include "present.h"

void copyKey(present_key_t *from, present_key_t *to) {
    for (int i = 0; i < PRESENT_80_KEY_SIZE_BYTES; i++) {
        (*to)[i] = (*from)[i];
    }
}

void copyBlock(block_t *from, block_t *to) {
    for (int i = 0; i < PRESENT_BLOCK_SIZE_BYTES; i++) {
        (*to)[i] = (*from)[i];
    }
}

void addRoundKey(block_t *block, round_key_t *roundKey) {
    for (int i = 0; i < ROUND_KEY_SIZE_BYTES; i++) {
        (*block)[i] ^= (*roundKey)[i];
    }
}

void pLayer(block_t *block) {
    block_t tempBlock = {0};
    for (int i = 0; i < 64; i++) {
        int byteIndex = i / 8;
        int bitIndex = i % 8;
        unsigned char bit = ((*block)[byteIndex] >> bitIndex) & 0x01;

        int permIndex = (16 * i) % 63;
        if (i == 63) {
            permIndex = 63;
        }

        int permByteIndex = permIndex / 8;
        int permBitIndex = permIndex % 8;
        tempBlock[permByteIndex] |= (bit << permBitIndex);
    }
    copyBlock(&tempBlock, block);
}

void generateRoundKeys80(present_key_t *suppliedKey, keys_t *keys) {
    present_key_t tempKey;
    copyKey(suppliedKey, &tempKey);

    for (int round = 0; round < ROUNDS; round++) {
        // Copy current key to round key (first 64 bits)
        for (int i = 0; i < 8; i++) {
            (*keys)[round][i] = tempKey[i];
        }

        // Rotate key 19 bits to the left
        present_key_t rotatedKey = {0};
        for (int i = 0; i < 80; i++) {
            int newBitPos = (i + 19) % 80;
            int byteIndex = i / 8;
            int bitIndex = i % 8;
            int newByteIndex = newBitPos / 8;
            int newBitIndex = newBitPos % 8;
            unsigned char bit = (tempKey[byteIndex] >> bitIndex) & 0x01;
            rotatedKey[newByteIndex] |= (bit << newBitIndex);
        }

        // Apply S-box to bits 79-76 (nibble 19) of rotated key
        unsigned char nibble = rotatedKey[9] >> 4;
        rotatedKey[9] = (rotatedKey[9] & 0x0F) | (sBox[nibble] << 4);

        // XOR with round counter at bits 19-15 (bits 19-15 are in byte 2, bits 7-4)
        rotatedKey[2] ^= (round + 1) << 7;
        rotatedKey[1] ^= (round + 1) >> 1;

        // Copy rotated key back to tempKey
        copyKey(&rotatedKey, &tempKey);
    }
}

void present80_encryptBlock(block_t *block, present_key_t *key) {
    keys_t roundKeys;
    generateRoundKeys80(key, &roundKeys);

    // Add initial round key (round 0)
    addRoundKey(block, &roundKeys[0]);

    // Perform 31 rounds
    for (int round = 1; round <= 31; round++) {
        // Substitution layer
        for (int i = 0; i < 8; i++) {
            unsigned char byte = (*block)[i];
            unsigned char lowNibble = byte & 0x0F;
            unsigned char highNibble = (byte >> 4) & 0x0F;
            (*block)[i] = (sBox[highNibble] << 4) | sBox[lowNibble];
        }

        // Permutation layer
        pLayer(block);

        // Add round key
        addRoundKey(block, &roundKeys[round]);
    }

    // Final round (32nd round) without permutation
    // Substitution layer
    for (int i = 0; i < 8; i++) {
        unsigned char byte = (*block)[i];
        unsigned char lowNibble = byte & 0x0F;
        unsigned char highNibble = (byte >> 4) & 0x0F;
        (*block)[i] = (sBox[highNibble] << 4) | sBox[lowNibble];
    }

    // Add final round key (round 32)
    addRoundKey(block, &roundKeys[32]);
}
