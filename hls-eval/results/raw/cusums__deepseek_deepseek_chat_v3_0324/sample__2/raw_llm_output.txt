    int S = 0;

    int inf = 0;
    
    for (int k = 0; k < N; k++) {
        if (epsilon[k] == 1) {
            S++;
        } else {
            S--;
        }
        
        if (S > sup) {
            sup = S;
        }
        if (S < inf) {
            inf = S;
        }
    }
    
    *res_sup = sup;
    *res_inf = inf;
}
